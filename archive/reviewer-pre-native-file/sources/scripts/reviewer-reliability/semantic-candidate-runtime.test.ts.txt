import { expect, test } from 'bun:test'
import { readFile } from 'node:fs/promises'
import { bindCandidateLifecycle, createNativeToolReviewTask } from './semantic-candidate-runtime'
import { createSemanticPromptCandidate } from './semantic-prompt-candidate'
import { screeningInput } from './fixtures'

test('process-only task factory returns the approved candidate without schema or admission changes', () => {
  const input = screeningInput()
  const candidate = createSemanticPromptCandidate(input, [])
  const task = createNativeToolReviewTask(input, [])
  expect(task.prompt).toBe(candidate.task.prompt)
  expect(task.schema).toEqual(candidate.task.schema)
  expect(Object.keys(task)).toEqual(Object.keys(candidate.task))
})
test('lifecycle transform changes only the task import and rejects drift or duplicates', async () => {
  const source = await readFile(new URL('../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-lifecycle.ts', import.meta.url), 'utf8')
  const result = bindCandidateLifecycle(source)
  const before = source.split('\n'), after = result.split('\n')
  expect(after.length).toBe(before.length)
  expect(after.filter((line, index) => line !== before[index])).toHaveLength(1)
  expect(() => bindCandidateLifecycle(result)).toThrow('binding drifted')
  expect(() => bindCandidateLifecycle(source + source)).toThrow('binding drifted')
})
test('production entry completes using candidate prompt in a synthetic native subprocess', async () => {
  const child = Bun.spawn([process.execPath, new URL('./semantic-candidate-probe.ts', import.meta.url).pathname], {
    env: { ...process.env, BEEGAME_REVIEWER_PROTOCOL: 'v2' }, stdout: 'pipe', stderr: 'pipe' })
  const [out, err, status] = await Promise.all([new Response(child.stdout).text(), new Response(child.stderr).text(), child.exited])
  expect({ status, err }).toEqual({ status: 0, err: '' })
  const summary = JSON.parse(out.trim().split('\n').at(-1)!)
  expect(summary.status).toBe('offline_candidate_lifecycle_passed')
  expect(summary.modelCalls).toBe(0)
  expect(summary.attempts).toBe(1)
}, 30_000)
