/** Frozen-response regression only. No network, credentials, model client or semantic auto-grader. */
import assert from 'node:assert/strict'
import { readFile } from 'node:fs/promises'
import { createHash } from 'node:crypto'
import { join } from 'node:path'
import { createNativeToolReviewTask } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-tool-task'
import { presentReviewInput } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-input'
import { reviewJson } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-contract'
import { verifyEvaluationTerminal } from './evaluation-finalization'

export async function reproduceSemanticReport(root: string) {
  const json = async (path: string) => JSON.parse(await readFile(join(root, path), 'utf8'))
  const manifest = await json('evidence-manifest.json')
  for (const entry of manifest.files) {
    assert(!entry.path.startsWith('/') && !entry.path.split('/').includes('..'), 'invalid_archive_path')
    const bytes = await readFile(join(root, entry.path))
    assert.equal(bytes.length, entry.bytes)
    assert.equal(createHash('sha256').update(bytes).digest('hex'), entry.sha256, `Changed evidence: ${entry.path}`)
  }
  const matching = (suffix: string) => {
    const found = manifest.files.filter((value: any) => value.path.endsWith(suffix))
    assert.equal(found.length, 1, 'Expected a unique frozen artifact')
    return found[0].path
  }
  const input = await json('frozen-input.json'), base = await json('base-task.json')
  const state = await json('case-1/state.json'), result = await json('case-1/result.json')
  const request = await json(matching('http-1.request.raw.json'))
  const texts = request.messages.flatMap((message: any) => typeof message.content === 'string'
    ? [message.content] : message.content.filter((block: any) => block.type === 'text').map((block: any) => block.text))
  assert(texts.some((text: string) => text.includes(base.prompt)), 'Original prompt was not delivered intact')
  const task = createNativeToolReviewTask(input, [])
  assert.equal(task.prompt, base.prompt, 'Candidate prompt changed')
  assert.deepEqual(task.schema, base.schema, 'Candidate report schema changed')
  assert.equal(presentReviewInput(input).paged, false)
  const blocks = new Map<number, { name: string; raw: string }>()
  for (const line of (await readFile(join(root, matching('http-2.response.raw.sse')), 'utf8')).split(/\r?\n/)) {
    if (!line.startsWith('data: ')) continue
    const event = JSON.parse(line.slice(6))
    if (event.type === 'content_block_start' && event.content_block?.type === 'tool_use')
      blocks.set(event.index, { name: event.content_block.name, raw: '' })
    if (event.type === 'content_block_delta' && event.delta?.type === 'input_json_delta')
      blocks.get(event.index)!.raw += event.delta.partial_json
  }
  const finals = [...blocks.values()].filter(block => block.name === 'StructuredOutput')
  assert.equal(finals.length, 1)
  const original = JSON.parse(finals[0]!.raw)
  const replay = task.admitWithTools(original, state.transcript.filter((entry: any) => entry.action.action === 'tools'))
  assert.equal(replay.kind, 'accepted')
  if (replay.kind !== 'accepted') throw new Error('Expected the historical admission boundary')
  assert.equal(reviewJson(replay.action.report), reviewJson(state.report))

  // The following labels are frozen regression expectations from the manual assessment,
  // NOT a general-purpose semantic detector and NEVER input to a live model prompt.
  const s08 = original.findings.findIndex((finding: any) => finding.title.startsWith('S08:'))
  assert(s08 >= 0)
  assert.deepEqual(original.findings[s08].evidence[0], { path: 'docs/UI_UX_SPEC.md', startLine: 8, endLine: 10 })
  const ui = input.documents.find((document: any) => document.path === 'docs/UI_UX_SPEC.md').content.split('\n')
  assert(ui.slice(7, 10).join('\n').includes('## S13'))
  assert(ui[12].includes('闪烁时推进战斗计时 200ms'))
  const invalid = structuredClone(original)
  invalid.findings[s08].evidence[0].endLine = 999
  const invalidDecision = task.admitWithTools(invalid, [])
  assert.equal(invalidDecision.kind, 'rejected')
  if (invalidDecision.kind === 'rejected') assert.equal(invalidDecision.diagnostic.reason, 'citation_mismatch')
  const correctedCitation = structuredClone(original)
  correctedCitation.findings[s08].evidence[0].startLine = 13
  correctedCitation.findings[s08].evidence[0].endLine = 13
  const correctedDecision = task.admitWithTools(correctedCitation, [])
  assert.equal(correctedDecision.kind, 'accepted')
  assert.equal(correctedCitation.findings.length, 7, 'Citation correction must not manufacture a missing finding')
  const asset = input.documents.find((document: any) => document.path === 'docs/ASSET_PLAN.md').content.split('\n')
  assert(asset[3].includes('status=planned') && asset[3].includes('res_glow 为 verified'))
  assert(original.summary.includes('S06') && original.summary.includes('are internally and cross-document consistent'))
  assert(!original.findings.some((finding: any) => finding.title.startsWith('S06:')))
  const metering = verifyEvaluationTerminal(await json('case-1/start.json'), { state, result, leaseStatus: 'free' })
  // 'free' here tests the pure verifier against a historical snapshot; it is NOT new live cleanup proof.
  return {
    status: 'historical_errors_reproduced', networkCalls: 0, modelCalls: 0,
    frozenFilesVerified: manifest.files.length, originalReportReplayedExactly: true,
    originalCitation: { rangeValid: true, exactQuote: true, relevance: 'manual_assessment_failed' },
    outOfBoundsCitationRejected: true, citationOnlyCorrectionStillHasSevenFindings: true,
    statusConflictDeliveredInline: true, missingStatusFindingPreserved: true,
    knownCause: 'Wrong judgment and line selection already present in raw provider report; not host transport corruption',
    causeHypothesis: 'Possible confusion between independent maturity/status fields and conflicting values of the same status; not proven from this single output',
    meteringReplay: metering, semanticQuality: 'historical_failure_unchanged',
  }
}

if (import.meta.main) {
  assert(process.argv.length === 3, 'Provide the immutable evidence directory')
  console.log(JSON.stringify(await reproduceSemanticReport(process.argv[2]!), null, 2))
}
