/** Offline subprocess probe; synthetic worker and billing, no network or credentials. */
import assert from 'node:assert/strict'
import { mkdtemp, readFile, writeFile, realpath } from 'node:fs/promises'
import { createHash } from 'node:crypto'
import { join } from 'node:path'
import { screeningInput } from './fixtures'
import { createSemanticPromptCandidate } from './semantic-prompt-candidate'
import { installSemanticCandidateRuntime } from './semantic-candidate-runtime'
import { observeEvaluation, verifyEvaluationTerminal } from './evaluation-finalization'
import type { UsageBillingInput } from '../../packages/agent-workflow-server/src/billing/public'
const input = screeningInput()
const expected = createSemanticPromptCandidate(input, [])
const sha = (v: string) => createHash('sha256').update(v).digest('hex')

if (process.argv[2] === '--worker') {
  const { createInterface } = await import('node:readline')
  for await (const line of createInterface({ input: process.stdin })) {
    const request = JSON.parse(line)
    if (request.type !== 'user') continue
    const prompt = request.message.content[0].text as string
    assert(prompt.startsWith(expected.task.prompt + '\nHost budget policy: '))
    await writeFile(join(process.cwd(), 'probe-prompt.json'), JSON.stringify({ candidatePromptSha256: sha(prompt.slice(0, expected.task.prompt.length)), fullPromptSha256: sha(prompt) }), { flag: 'wx' })
    const report = { status: 'completed', summary: 'Offline transport probe, not a semantic assessment',
      reviewedDocuments: input.documents.map(d => d.path), requirementsAssessment: 'reviewed', findings: [],
      omissionCheck: 'Synthetic result only', limitations: ['No model was called'] }
    console.log(JSON.stringify({ type: 'assistant', message: { id: 'synthetic-request', usage: { input_tokens: 20, output_tokens: 10 }, content: [] } }))
    console.log(JSON.stringify({ type: 'result', subtype: 'success', structured_output: report,
      modelUsage: { fixture: { input_tokens: 20, output_tokens: 10 } } }))
  }
} else {
  const repo = join(import.meta.dir, '../..')
  const lifecycle = join(repo, 'packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-lifecycle.ts')
  const binding = installSemanticCandidateRuntime(sha(await readFile(lifecycle, 'utf8')))
  const { createFoundationReviewer, foundationReviewerInputDigest } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/public.ts')
  const { openSingleReviewJournal } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-journal.ts')
  binding.assertBound()
  const workspace = await realpath(await mkdtemp('/private/tmp/reviewer-r1-offline-probe-'))
  let billingCalls = 0
  const stage = createFoundationReviewer({ workspacePath: workspace }, {
    runtimeConfiguration: { protocolVersion: 1, ownerId: 'offline-owner', resolve: async () => ({ command: [process.execPath, import.meta.path, '--worker'], env: {} }) },
    billing: { protocolVersion: 1, ownerId: 'offline-owner', async submitUsage(value: UsageBillingInput) {
      billingCalls++
      return { protocolVersion: 1, ownerId: value.ownerId, eventId: value.eventId, idempotencyKey: value.eventId,
        usage: value.usage, pricingVersion: 'weighted-v1', usageSource: 'runtime_snapshot', duplicate: false,
        recorded: { eventId: value.eventId, duplicate: false, weightedTokensDelta: 70, creditsMicro: 7000 },
        debited: { eventId: value.eventId, duplicate: false, weightedTokensDelta: 70, creditsMicro: 7000 } }
    } },
  })
  const start = { protocolVersion: 1 as const, ownerId: 'offline-owner', projectId: 'offline-project', requestId: 'offline-probe', input,
    inputDigest: foundationReviewerInputDigest({ ownerId: 'offline-owner', projectId: 'offline-project', input }) }
  const handle = await stage.start(start)
  const observation = await observeEvaluation({ async read() {
    const journal = await openSingleReviewJournal(workspace, handle.id, start, true)
    return { result: await stage.getResult(handle), state: await journal.read(), leaseStatus: await journal.leaseStatus() }
  }, stop: reason => stage.stop({ handle, requestId: 'probe-stop', reason }), save: async () => {},
    creditLimit: 12, runTimeoutMs: 15_000, drainTimeoutMs: 5000, pollMs: 50 })
  assert(observation.snapshot)
  const verified = verifyEvaluationTerminal(start, observation.snapshot)
  assert.equal(observation.snapshot.result.status, 'completed')
  assert.equal(observation.snapshot.state.attempts.length, 1)
  assert.equal(billingCalls, 1)
  const markers: any[] = []
  for await (const path of new Bun.Glob('**/probe-prompt.json').scan({ cwd: workspace, absolute: true, dot: true }))
    markers.push(JSON.parse(await readFile(path, 'utf8')))
  assert.equal(markers.length, 1)
  assert.equal(markers[0].candidatePromptSha256, expected.candidatePromptSha256)
  console.log(JSON.stringify({ status: 'offline_candidate_lifecycle_passed', modelCalls: 0, networkCalls: 0,
    attempts: 1, syntheticReceiptCreditsMicro: verified.creditsMicro, markers, workspace }))
}
