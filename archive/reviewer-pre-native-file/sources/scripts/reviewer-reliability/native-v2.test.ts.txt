import { expect, test } from 'bun:test'
import { fixtureInputs } from './fixtures'
import { checkNativeFixtures, exportNativeFixture } from './native-v2-check'

test('frozen screening and reserve sources are ready for v2, without claiming semantic acceptance', async () => {
  const result = await checkNativeFixtures()
  expect(result.status).toBe('offline_ready')
  expect(result.semanticQuality).toBe('not_evaluated')
  expect(result.modelCalls).toBe(0)
  expect(result.fixtures.map(value => value.id)).toEqual(['B24', 'R41', 'R86'])
  expect(result.fixtures.every(value => value.sources === 9 && value.toolCalls === 9)).toBe(true)
})

test('input export preserves source exactly and has no scoring payload', () => {
  for (const [id, input] of Object.entries(fixtureInputs())) {
    const exported = exportNativeFixture(id)
    expect(exported).toEqual(input)
    expect(Object.keys(exported).sort()).toEqual(['documents', 'requirements', 'subject'])
  }
  expect(() => exportNativeFixture('toString')).toThrow()
  expect(() => exportNativeFixture('unknown')).toThrow()
})
