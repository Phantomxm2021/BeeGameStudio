// Offline fixture readiness only: no model client, credentials, or semantic auto-grading.
import assert from 'node:assert/strict'
import { createHash } from 'node:crypto'
import { mkdtemp, realpath, rm } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { fixtureInputs, minimalCases, pairs } from './fixtures'
import rubric from './rubric.json'
import { presentReviewInput } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-input'
import { createNativeToolReviewTask } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-tool-task'
import { executeNativeReviewTool, prepareReviewToolSession, readReviewToolRecords, reviewNativeToolSchemas } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-native-tools'

const sha = (value: string) => createHash('sha256').update(value).digest('hex')
type FixtureId = keyof ReturnType<typeof fixtureInputs>

export function exportNativeFixture(id: string) {
  const inputs = fixtureInputs()
  assert(Object.hasOwn(inputs, id), 'Use B24, R41 or R86')
  // Deliberately return only source input, never an answer, rubric or previous report.
  return inputs[id as FixtureId]
}

export async function checkNativeFixtures() {
  const inputs = fixtureInputs()
  const cases = minimalCases()
  assert.equal(cases.length, 16)
  assert.equal(new Set(cases.map(value => value.id)).size, 16)
  for (const [index, pair] of pairs.entries()) {
    const [normal, defect] = pair.ids.map(id => cases.find(value => value.id === id)!)
    assert.equal(normal!.documents.filter((value, i) => value.text !== defect!.documents[i]!.text).length, 1)
    assert.equal(rubric.pairs[index]!.normal, pair.ids[0])
    assert.equal(rubric.pairs[index]!.defect, pair.ids[1])
  }
  assert.deepEqual(inputs.R41.documents.filter((value, i) => value.content !== inputs.R86.documents[i]!.content)
    .map(value => value.path), ['docs/GDD.md', 'docs/LEVEL_SCENE_DESIGN.md', 'docs/BALANCE_DESIGN.md'])
  const root = await realpath(await mkdtemp(join(tmpdir(), 'review-v2-fixtures-')))
  const fixtures = []
  try {
    for (const id of Object.keys(inputs) as FixtureId[]) {
      const input = inputs[id]
      const before = JSON.stringify(input)
      assert.equal(sha(before), rubric.fixtures[id], `Frozen fixture changed: ${id}`)
      const task = createNativeToolReviewTask(input, [])
      assert.equal(task.format, 'foundation-review-tools-v2')
      const schemas = [...reviewNativeToolSchemas().map(value => value.input_schema), task.schema]
      assert.equal(schemas.length, 10)
      for (const schema of schemas) {
        assert.equal(schema.type, 'object')
        assert(!/"(?:oneOf|anyOf|\$ref|payload_json|numericEvidence|numericBindings)"/.test(JSON.stringify(schema)))
      }
      const { paged, source } = presentReviewInput(input)
      assert.equal(paged, false, 'Frozen small fixtures should fit inline')
      const sourceLine = task.prompt.split('INPUT (each text field is one exact original line, excluding its newline; line is its one-based source position):\n')[1]?.split('\n')[0]
      assert(sourceLine, 'Native prompt must contain the delivered input')
      assert.deepEqual(JSON.parse(sourceLine), source)
      assert(!task.prompt.includes(rubric.scoring))
      for (const pair of rubric.pairs) assert(!task.prompt.includes(pair.judgment))
      const descriptor = await prepareReviewToolSession(join(root, id), id, input, [])
      const sources = [...input.documents, ...(input.requirements.status === 'available'
        ? [{ path: 'requirements', content: input.requirements.content }] : [])]
      let toolCalls = 0
      for (const original of sources) {
        const lines = original.content.split('\n')
        const delivered: string[] = []
        for (let startLine = 1; startLine <= lines.length; startLine += 200) {
          const endLine = Math.min(startLine + 199, lines.length)
          const feedback = await executeNativeReviewTool(descriptor, 'review_read', { path: original.path, startLine, endLine }, `read-${++toolCalls}`) as { result: { content: string } }[]
          assert.equal(feedback.length, 1)
          delivered.push(feedback[0]!.result.content)
        }
        assert.equal(delivered.join('\n'), original.content, `Native read changed source: ${id}/${original.path}`)
      }
      assert.equal((await readReviewToolRecords(descriptor)).length, toolCalls)
      assert.equal(JSON.stringify(input), before, 'Readiness checks must not change frozen input')
      fixtures.push({ id, sha256: sha(before), bytes: Buffer.byteLength(before), sources: sources.length, toolCalls })
    }
  } finally {
    // Only this invocation's generated tool journals; never fixture or historical data.
    await rm(root, { recursive: true, force: true })
  }
  return { status: 'offline_ready', protocol: 'foundation-review-tools-v2', semanticQuality: 'not_evaluated',
    modelCalls: 0, fixtures }
}

if (import.meta.main) {
  const args = process.argv.slice(2)
  if (args.length === 2 && args[0] === '--input') console.log(JSON.stringify(exportNativeFixture(args[1]!)))
  else {
    assert.equal(args.length, 0, 'Use no arguments or --input B24|R41|R86; no model-run option exists')
    console.log(JSON.stringify(await checkNativeFixtures(), null, 2))
  }
}
