/** Test-runner adapter, not a second authentication implementation.
 * The existing local session routes own refresh-token rotation and persistence.
 * This adapter pins one owner/session, calls those routes, then rereads the store.
 * No refresh token, request header, or upstream error body is returned as evidence.
 */
export type EvaluationSession = {
  id: string; ownerId: string; accessToken: string; expiresAt: number; sessionExpiresAt: number
}
export type EvaluationCredential = {
  getValidAccessToken(options?: { forceRefresh?: boolean }): Promise<string>
}

export function createEvaluationCredential(options: {
  ownerId: string
  runtimeOrigin: string
  frontendOrigin: string
  loadSessions(): Promise<EvaluationSession[]>
  fetchImpl: typeof fetch
  now?: () => number
}): EvaluationCredential {
  const runtime = new URL(options.runtimeOrigin)
  const frontend = new URL(options.frontendOrigin)
  for (const url of [runtime, frontend]) {
    if (url.protocol !== 'http:' || url.hostname !== '127.0.0.1' || url.pathname !== '/' ||
      url.username || url.password || url.search || url.hash) throw new Error('evaluation_auth_origin_invalid')
  }
  const now = options.now ?? Date.now
  let sessionId: string | undefined
  let pending: Promise<string> | undefined
  const fresh = (record: EvaluationSession) => record.expiresAt > now() + 30_000

  async function record(): Promise<EvaluationSession> {
    const sessions = await options.loadSessions()
    const candidates = sessions.filter(value => value.ownerId === options.ownerId &&
      /^[A-Za-z0-9_-]+$/.test(value.id) && typeof value.accessToken === 'string' && value.accessToken.length > 0 &&
      Number.isFinite(value.expiresAt) && Number.isFinite(value.sessionExpiresAt) && value.sessionExpiresAt > now())
      .sort((a, b) => b.expiresAt - a.expiresAt)
    // Once selected, never silently switch session or account after logout/expiry.
    const selected = sessionId ? candidates.find(value => value.id === sessionId) : candidates[0]
    if (!selected) throw new Error('evaluation_auth_session_unavailable')
    sessionId = selected.id
    return selected
  }

  async function refresh(current: EvaluationSession): Promise<string> {
    const headers = { cookie: `beegame_session=${current.id}`, origin: frontend.origin }
    for (const [path, method] of [['/api/auth/session', 'GET'], ['/api/auth/session/refresh', 'POST']] as const) {
      const response = await options.fetchImpl(new URL(path, runtime), {
        method, headers, redirect: 'error', signal: AbortSignal.timeout(20_000),
      })
      if (!response.ok) throw new Error('evaluation_auth_refresh_unavailable')
      const body = await response.json() as { authenticated?: boolean; user?: { id?: string } }
      if (body.authenticated !== true || body.user?.id !== options.ownerId)
        throw new Error('evaluation_auth_owner_mismatch')
    }
    const renewed = await record()
    if (!fresh(renewed)) throw new Error('evaluation_auth_refresh_not_persisted')
    return renewed.accessToken
  }

  return Object.freeze({
    async getValidAccessToken(request?: { forceRefresh?: boolean }) {
      // Share forced refreshes as well as expiry refreshes within the runner.
      // The local server also serializes refreshes for the same session cookie.
      if (pending) return pending
      const operation = (async () => {
        try {
          const current = await record()
          return !request?.forceRefresh && fresh(current) ? current.accessToken : await refresh(current)
        } catch {
          // Never let tokens, headers or upstream bodies escape through errors.
          throw new Error('evaluation_auth_unavailable')
        }
      })()
      pending = operation
      try { return await operation } finally { if (pending === operation) pending = undefined }
    },
  })
}

/** Only read-only calls may use this bounded 401 retry. Never wrap billing or model calls. */
export async function authenticatedEvaluationRead<T>(credential: EvaluationCredential,
  read: (token: string) => Promise<T>): Promise<T> {
  try { return await read(await credential.getValidAccessToken()) }
  catch (error) {
    // The existing dashboard store exposes REST status in this exact prefix,
    // whereas the runtime-env client uses a typed status. Do not match arbitrary
    // body text containing "401", and never surface either error's body.
    const unauthorized = (error as { status?: unknown })?.status === 401 ||
      (error instanceof Error && /^Supabase request failed: 401(?:\s|$)/u.test(error.message))
    if (!unauthorized) throw new Error('evaluation_read_unavailable')
  }
  try { return await read(await credential.getValidAccessToken({ forceRefresh: true })) }
  catch { throw new Error('evaluation_read_unavailable') }
}
