import assert from 'node:assert/strict'
import { reviewDigest } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-contract'
import { buildSingleReviewResult, type SingleReviewStart, type SingleReviewResult } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-publication'
import type { SingleReviewState } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-journal'

export type EvaluationSnapshot = { result: SingleReviewResult; state: SingleReviewState; leaseStatus: 'free' | 'busy' | 'stale' }
type Usage = { prompt_tokens: number; completion_tokens: number; cache_read_tokens: number; cache_creation_tokens: number; total_tokens: number }
export function evaluationUsage(state: SingleReviewState): Usage {
  const total: Usage = { prompt_tokens: 0, completion_tokens: 0, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 0 }
  for (const attempt of state.attempts) {
    const usage = attempt.closed?.usage ?? attempt.usage
    if (!usage) continue // Running attempts may not have emitted a usage frame yet.
    for (const key of Object.keys(total) as (keyof Usage)[]) {
      assert(Number.isSafeInteger(usage[key]) && usage[key] >= 0, 'invalid_usage')
      total[key] += usage[key]
      assert(Number.isSafeInteger(total[key]), 'usage_overflow')
    }
    assert.equal(usage.total_tokens, usage.prompt_tokens + usage.completion_tokens + usage.cache_read_tokens + usage.cache_creation_tokens, 'invalid_usage_total')
  }
  return total
}
export function evaluationCredits(usage: Usage): number {
  return Math.ceil((100 * usage.prompt_tokens + 10 * usage.cache_read_tokens + 125 * usage.cache_creation_tokens + 500 * usage.completion_tokens) / 100) / 10_000
}

/** Structural / metering checks only. This intentionally makes NO semantic-quality claim. */
export function verifyEvaluationTerminal(start: SingleReviewStart, snapshot: EvaluationSnapshot) {
  const { result, state, leaseStatus } = snapshot
  assert.notEqual(result.status, 'running', 'lifecycle_still_running')
  assert.equal(leaseStatus, 'free', 'execution_cleanup_unverified')
  assert.equal(result.billingStatus, 'settled', 'billing_unsettled')
  const attempts = new Set<string>(), events = new Set<string>()
  let creditsMicro = 0
  for (const attempt of state.attempts) {
    assert(!attempts.has(attempt.id), 'duplicate_attempt')
    attempts.add(attempt.id)
    assert(attempt.closed?.usageObserved, 'usage_not_sealed')
    assert.equal(attempt.closed.attemptId, attempt.id, 'seal_attempt_mismatch')
    const receipt = attempt.receipt as any
    assert(receipt, 'receipt_missing')
    assert.equal(receipt.protocolVersion, 1, 'receipt_protocol')
    assert.equal(receipt.ownerId, start.ownerId, 'receipt_owner')
    const eventId = reviewDigest({ producer: 'foundation-reviewer', handle: result.handle.id, attemptId: attempt.id, usage: attempt.closed.usage })
    assert.equal(receipt.eventId, eventId, 'receipt_event')
    assert.equal(receipt.idempotencyKey, eventId, 'receipt_idempotency')
    assert(!events.has(eventId), 'duplicate_receipt')
    events.add(eventId)
    assert.deepEqual(receipt.usage, attempt.closed.usage, 'receipt_usage')
    assert.equal(receipt.pricingVersion, 'weighted-v1', 'receipt_pricing')
    assert.equal(receipt.usageSource, 'runtime_snapshot', 'receipt_source')
    if ('skipped' in receipt) {
      assert.equal(receipt.skipped, true, 'invalid_skipped_receipt')
      assert.equal(receipt.usage.total_tokens, 0, 'nonzero_skipped_usage')
    } else {
      assert(typeof receipt.recorded?.eventId === 'string' && receipt.recorded.eventId.length > 0, 'record_event_missing')
      assert.equal(receipt.recorded.eventId, receipt.debited?.eventId, 'debit_event')
      const micro = Math.round(evaluationCredits(attempt.closed.usage) * 1e6)
      assert.equal(receipt.recorded.creditsMicro, micro, 'record_amount')
      assert.equal(receipt.debited.creditsMicro, micro, 'debit_amount')
      creditsMicro += micro
    }
  }
  if (result.status === 'completed') {
    assert(state.report && state.published && !state.failure && !state.stopped, 'publication_incomplete')
    const expected = buildSingleReviewResult(start, state.report)
    assert.equal(state.published, expected.resultId, 'publication_id')
    assert.deepEqual(result.result, expected, 'publication_binding')
  }
  return { creditsMicro, usage: evaluationUsage(state), meteringVerified: true as const, semanticQuality: 'not_evaluated' as const }
}

/** Public lifecycle observation; never starts, resumes or recreates a review.
 * A terminal label alone is not enough: wait for cleanup AND metering settlement.
 * Every error/stop consumes a separate bounded drain period, not the expired run deadline.
 */
export async function observeEvaluation(options: {
  read(): Promise<EvaluationSnapshot>
  stop(reason: 'credit_guard' | 'timeout' | 'runner_error'): Promise<unknown>
  save(snapshot: EvaluationSnapshot): Promise<void>
  creditLimit: number; runTimeoutMs: number; drainTimeoutMs: number
  now?: () => number; sleep?: (ms: number) => Promise<void>; pollMs?: number
}) {
  for (const value of [options.creditLimit, options.runTimeoutMs, options.drainTimeoutMs])
    assert(Number.isFinite(value) && value > 0, 'invalid_runner_limit')
  assert(Number.isFinite(options.pollMs ?? 500) && (options.pollMs ?? 500) > 0, 'invalid_poll_interval')
  const now = options.now ?? Date.now, sleep = options.sleep ?? (ms => Bun.sleep(ms))
  const runDeadline = now() + options.runTimeoutMs
  let drainDeadline: number | undefined
  let stopReason: 'credit_guard' | 'timeout' | 'runner_error' | undefined
  let stopFailed = false, observationFailed = false
  let lastSnapshot: EvaluationSnapshot | undefined
  async function requestStop(reason: typeof stopReason) {
    if (stopReason) return
    stopReason = reason
    drainDeadline = now() + options.drainTimeoutMs
    try { await options.stop(reason!) } catch { stopFailed = true }
  }
  for (;;) {
    try {
      lastSnapshot = await options.read()
      await options.save(lastSnapshot) // Keep full status/state evidence, including failures.
      const { result, state, leaseStatus } = lastSnapshot
      if (result.status !== 'running' && leaseStatus === 'free' && result.billingStatus === 'settled' &&
        state.attempts.every(attempt => attempt.closed?.usageObserved && attempt.receipt)) break
      if (result.status === 'running' && evaluationCredits(evaluationUsage(state)) >= options.creditLimit)
        await requestStop('credit_guard')
    } catch {
      observationFailed = true
      await requestStop('runner_error')
    }
    if (!stopReason && now() >= runDeadline) await requestStop('timeout')
    if (drainDeadline !== undefined && now() >= drainDeadline) break
    await sleep(options.pollMs ?? 500)
  }
  return { snapshot: lastSnapshot, stopRequested: stopReason !== undefined, stopReason,
    stopFailed, observationFailed, drainTimedOut: drainDeadline !== undefined && now() >= drainDeadline }
}

/** Unknown is not changed; neither unknown nor changed may authorize another paid case. */
export async function verifyEvaluationConfiguration<T>(before: T, readAfter: () => Promise<T>, timeoutMs = 30_000) {
  assert(Number.isFinite(timeoutMs) && timeoutMs > 0, 'invalid_configuration_timeout')
  let timer: ReturnType<typeof setTimeout> | undefined
  try {
    const after = await Promise.race([readAfter(), new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(new Error('configuration_timeout')), timeoutMs)
    })])
    return { status: reviewDigest(before) === reviewDigest(after) ? 'unchanged' as const : 'changed' as const, after }
  } catch { return { status: 'unknown' as const, error: 'configuration_unavailable' as const } }
  finally { if (timer) clearTimeout(timer) }
}
