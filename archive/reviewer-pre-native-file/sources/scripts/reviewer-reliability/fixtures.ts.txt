export function makeInput(code: string, requirements: string, excerpts: Excerpt[]) {
  const documents = REVIEW_DOCUMENT_PATHS.map(path => ({ path,
    content: `# ${path.slice(5, -3)}\n\n` + excerpts.filter(e => e.path === path).map(e => e.text).join('\n\n') }))
  const digest = reviewHash(requirements)
  return captureSingleReviewInput({ subject: { kind: 'initial', sourceResultId: `evaluation-${code}`,
    sourceDocumentRevision: reviewHash(JSON.stringify(documents)), confirmedBriefDigest: digest },
    requirements: { status: 'available', content: requirements, digest }, documents })
}

export function minimalCases() {
  return pairs.flatMap(pair => pair.ids.map((id, index) => ({ id,
    documents: pair.documents.map(d => ({ ...d, text: index === 1 && d.path === pair.change.path
      ? d.text.replace(pair.change.from, pair.change.to) : d.text })) })))
}
