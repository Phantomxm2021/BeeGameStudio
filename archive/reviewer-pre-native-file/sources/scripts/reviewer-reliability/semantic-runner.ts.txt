type Usage = {
  prompt_tokens: number
  completion_tokens: number
  cache_read_tokens: number
  cache_creation_tokens: number
  total_tokens: number
}

type ModelConfigSummary = {
  id: string
  ownerId: string
  name: string
  provider: string
  baseUrl?: string
  models: Record<string, string>
  isDefault: boolean
  createdAt: string
  updatedAt: string
}

const emptyUsage = (): Usage => ({ prompt_tokens: 0, completion_tokens: 0,
  cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 0 })

function addUsage(target: Usage, value: Usage): void {
  for (const key of Object.keys(target) as (keyof Usage)[]) target[key] += value[key]
}

function sha(value: string | Buffer): string {
  return createHash('sha256').update(value).digest('hex')
}

function jsonClone<T>(value: T): T {
  return JSON.parse(JSON.stringify(value)) as T
}

async function readJson<T>(path: string): Promise<T> {
  return JSON.parse(await readFile(path, 'utf8')) as T
}

async function writeSnapshot(path: string, value: unknown): Promise<void> {
  await mkdir(dirname(path), { recursive: true })
  await writeFile(path, `${JSON.stringify(value, null, 2)}\n`, { mode: 0o600 })
}

async function writeImmutable(path: string, value: unknown): Promise<void> {
  await mkdir(dirname(path), { recursive: true })
  try {
    const file = await open(path, 'wx', 0o600)
    try { await file.writeFile(`${JSON.stringify(value, null, 2)}\n`); await file.sync() }
    finally { await file.close() }
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error
    assert.deepEqual(await readJson(path), value, `immutable evidence conflict: ${path}`)
  }
}

async function runFreeze(): Promise<Record<string, unknown>> {
  const child = Bun.spawn([process.execPath, freezePath], { stdout: 'pipe', stderr: 'pipe' })
  const [stdout, stderr, exitCode] = await Promise.all([
    new Response(child.stdout).text(),
    new Response(child.stderr).text(),
    child.exited,
  ])
  if (exitCode !== 0) {
    const detail = `${stdout}\n${stderr}`.trim().replace(/\s+/g, ' ').slice(0, 500)
    throw new Error(`Frozen input/source verification failed${detail ? `: ${detail}` : ''}`)
  }
  const line = stdout.trim().split(/\r?\n/).at(-1) ?? ''
  return JSON.parse(line) as Record<string, unknown>
}

function parseMode(): { kind: 'preflight' } | { kind: 'execute'; caseId: 'case-1' | 'case-2' } {
  const args = process.argv.slice(2)
  if (args.length === 0 || (args.length === 1 && args[0] === '--preflight')) return { kind: 'preflight' }
  let selected: string | undefined
  if (args.length === 2 && args[0] === '--execute-once') selected = args[1]
  else if (args.length === 1 && args[0]?.startsWith('--execute-once=')) selected = args[0].slice('--execute-once='.length)
  else throw new Error('Use no arguments for preflight, or --execute-once case-1|case-2')
  const caseId = selected === 'case1' || selected === 'case-1' ? 'case-1'
    : selected === 'case2' || selected === 'case-2' ? 'case-2' : undefined
  if (!caseId) throw new Error('Paid execution requires case-1 or case-2')
  return { kind: 'execute', caseId }
}

function normalizeUsage(value: Usage): Usage {
  const usage = jsonClone(value)
  for (const key of Object.keys(emptyUsage()) as (keyof Usage)[]) {
    assert(Number.isSafeInteger(usage[key]) && usage[key] >= 0, `invalid usage ${key}`)
  }
  assert.equal(usage.total_tokens, usage.prompt_tokens + usage.completion_tokens + usage.cache_read_tokens + usage.cache_creation_tokens)
  return usage
}

/** Same weighted-v1 estimate used by the previous live runner. */
function weightedV1Credits(usage: Usage): number {
  const weighted = 100 * usage.prompt_tokens + 10 * usage.cache_read_tokens +
    125 * usage.cache_creation_tokens + 500 * usage.completion_tokens
  return Math.ceil(weighted / 100) / 10_000
}

function summarizeModelConfigs(configs: readonly any[]): { configs: ModelConfigSummary[]; digest: string } {
  const safe = configs.map(config => ({
    id: String(config.id), ownerId: String(config.ownerId), name: String(config.name),
    provider: String(config.provider), ...(config.baseUrl ? { baseUrl: String(config.baseUrl) } : {}),
    models: Object.fromEntries(Object.entries(config.models ?? {}).filter(([key, value]) =>
      ['fast', 'balanced', 'strong'].includes(key) && typeof value === 'string')) as Record<string, string>,
    isDefault: Boolean(config.isDefault),
    createdAt: new Date(config.createdAt).toISOString(), updatedAt: new Date(config.updatedAt).toISOString(),
  })).sort((a, b) => a.id.localeCompare(b.id))
  return { configs: safe, digest: sha(JSON.stringify(safe)) }
}

function stateUsage(state: any): Usage {
  const total = emptyUsage()
  for (const attempt of state.attempts ?? []) {
    const usage = attempt.closed?.usage ?? attempt.usage
    if (usage) addUsage(total, normalizeUsage(usage))
  }
  return total
}

function receiptCreditsMicro(state: any): number {
  let creditsMicro = 0
  for (const attempt of state.attempts ?? []) {
    const receipt = attempt.receipt
    if (!receipt || receipt.skipped) continue
    assert.equal(receipt.recorded.eventId, receipt.debited.eventId)
    assert.equal(receipt.recorded.creditsMicro, receipt.debited.creditsMicro)
    creditsMicro += receipt.recorded.creditsMicro
  }
  return creditsMicro
}

const evidenceSkipDirectories = new Set(['config', 'node_modules', 'npm-cache', '.npm'])

async function evidenceFiles(directory: string): Promise<string[]> {
  const files: string[] = []
  for (const entry of await readdir(directory, { withFileTypes: true })) {
    if (entry.isDirectory() && evidenceSkipDirectories.has(entry.name)) continue
    const path = join(directory, entry.name)
    if (entry.isDirectory()) files.push(...await evidenceFiles(path))
    else if (entry.isFile()) files.push(path)
  }
  return files
}

async function scanEvidence(caseRoot: string, selectedEnv: Record<string, string>): Promise<{
  requestCount: number; credentialLeak: boolean; rawFiles: number; captureLimitHit: boolean
}> {
  const files = await evidenceFiles(caseRoot)
  const requestPaths = files.filter(path => /\/http-\d+\.request\.raw\.json$/u.test(path))
  const rawPaths = files.filter(path => /\.raw\.(?:json|sse)$/u.test(path))
  const captureLimitHit = files.some(path => path.endsWith('/capture-http-limit.json'))
  const credentialValues = Object.entries(selectedEnv)
    .filter(([key, value]) => /API_KEY|AUTH_TOKEN/u.test(key) && value.length > 8)
    .map(([, value]) => Buffer.from(value))
  let credentialLeak = false
  for (const path of rawPaths) {
    const bytes = await readFile(path)
    if (credentialValues.some(value => bytes.includes(value))) credentialLeak = true
  }
  return { requestCount: requestPaths.length, credentialLeak, rawFiles: rawPaths.length, captureLimitHit }
}

async function runCase(caseId: 'case-1' | 'case-2', input: any, base: any): Promise<void> {
  assert(registration && batchRoot, 'A new paid registration is required')
  const { createNativeToolReviewTask } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-tool-task.ts')
  const { reviewNativeToolSchemas } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-native-tools.ts')
  const { createSemanticPromptCandidate } = await import('./semantic-prompt-candidate')
  const candidate = createSemanticPromptCandidate(input, [])
  assert.equal(candidate.id, registration.candidateId)
  assert.equal(candidate.candidatePromptSha256, registration.candidatePromptSha256)
  assert.equal(candidate.baselinePromptSha256, registration.baselinePromptSha256)
  assert.equal(sha(createNativeToolReviewTask(input, []).prompt), registration.baselinePromptSha256)
  const expectedTask = candidate.task
  assert.equal(expectedTask.prompt, base.prompt, 'Frozen prompt mismatch')
  assert.deepEqual(expectedTask.schema, base.schema, 'Frozen report schema mismatch')
  assert.deepEqual(reviewNativeToolSchemas(), base.tools, 'Frozen tool schema mismatch')
  const caseRoot = join(batchRoot, caseId)
  await mkdir(caseRoot, { recursive: true })
  if ((await readdir(caseRoot)).includes('execute.claimed')) throw new Error(`${caseId} was already claimed; refusing retry`)
  if (caseId === 'case-2') {
    const first = await readJson<any>(join(batchRoot, 'case-1', 'completed.json')).catch(() => undefined)
    if (!first || first.status !== 'completed' || first.billingStatus !== 'settled' || first.unsettled || first.configUnchanged !== true ||
      first.integrityOk !== true || first.credentialLeak === true || first.captureLimitHit === true ||
      first.runtimeGuardHit === true || first.stopRequested === true || Number(first.requestCount) > 24)
      throw new Error('case-2 requires a settled, configuration-stable, integrity-safe case-1')
    if (batchReserveCredits - Number(first.creditsMicro ?? 0) / 1e6 < observedCaseGuardCredits)
      throw new Error('case-2 batch reserve is unavailable')
  }

  const manifest = await readJson<any>(join(batchRoot, 'frozen-manifest.json'))
  const lifecyclePath = repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-lifecycle.ts'
  const { installSemanticCandidateRuntime } = await import('./semantic-candidate-runtime')
  const candidateBinding = installSemanticCandidateRuntime(manifest.sourceHashes[lifecyclePath])
  const { createFoundationReviewer, foundationReviewerInputDigest } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/public.ts')
  candidateBinding.assertBound()
  await writeImmutable(join(caseRoot, 'candidate-binding.json'), { id: candidate.id,
    candidatePromptSha256: candidate.candidatePromptSha256, baselinePromptSha256: candidate.baselinePromptSha256,
    lifecycleSourceSha256: manifest.sourceHashes[lifecyclePath], processOnly: true })
  const { openSingleReviewJournal } = await import(repo + '/packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-journal.ts')
  const start: any = { protocolVersion: 1, ownerId, projectId,
    requestId: `${registration.batchId}-${caseId}`, input }
  start.inputDigest = foundationReviewerInputDigest(start)
  await writeImmutable(join(caseRoot, 'start.json'), start)
  const infrastructure = await makeInfrastructure(caseRoot)
  const wallet = await infrastructure.readWallet()
  assert(wallet && Number.isSafeInteger(wallet.balanceCreditsMicro) && wallet.balanceCreditsMicro >= (caseId === 'case-1' ? batchReserveCredits : observedCaseGuardCredits) * 1e6,
    'Account balance cannot cover the registered reserve')
  await writeImmutable(join(caseRoot, 'balance-preflight.json'), { sufficientForRegisteredReserve: true, modelCalls: 0 })
  const before = summarizeModelConfigs(await infrastructure.readModelConfigs())
  await writeImmutable(join(caseRoot, 'model-config-before.json'), before)
  const resolved = await infrastructure.configuration.resolve({ ownerId })
  assert.equal(resolved.env.ANTHROPIC_BASE_URL?.replace(/\/$/u, ''), originalArkBaseUrl)
  for (const key of modelEnvKeys) assert.equal(resolved.env[key], model)
  assert(resolved.command.includes('--model') && resolved.command.includes(model))
  await writeImmutable(join(caseRoot, 'selected-runtime.json'), safeSelectedConfiguration(resolved))

  if (process.env.BEEGAME_REVIEWER_PROTOCOL === 'legacy-v1')
    throw new Error('Refusing paid run with legacy reviewer protocol')
  process.env.BEEGAME_REVIEWER_PROTOCOL = 'v2'

  // This is the only execution claim.  It is intentionally created after all
  // zero-model configuration checks, and is never removed on any outcome.
  const claim = await open(join(caseRoot, 'execute.claimed'), 'wx', 0o600)
  try { await claim.writeFile(JSON.stringify({ case: caseId, model, claimedAt: new Date().toISOString() })); await claim.sync() }
  finally { await claim.close() }

  const stage = createFoundationReviewer({ workspacePath: caseRoot }, {
    runtimeConfiguration: infrastructure.configuration, billing: infrastructure.billing,
  })
  let handle: any
  let terminal: any
  let observation: Awaited<ReturnType<typeof observeEvaluation>> | undefined
  let lifecycleVerified = false
  let lifecycleError: string | undefined
  try {
    handle = await stage.start(start)
    let initialEvidenceFailed = false
    try { await writeImmutable(join(caseRoot, 'handle.json'), handle) }
    catch { initialEvidenceFailed = true }
    observation = await observeEvaluation({
      async read() {
        const result = await stage.getResult(handle)
        const journal = await openSingleReviewJournal(caseRoot, handle.id, start, true)
        return { result, state: await journal.read(), leaseStatus: await journal.leaseStatus() }
      },
      stop: reason => stage.stop({ handle, requestId: `runner-stop-${start.requestId}`, reason }),
      async save(snapshot) {
        if (initialEvidenceFailed) throw new Error('initial_evidence_write_failed')
        await writeSnapshot(join(caseRoot, 'state.json'), snapshot.state)
        await writeSnapshot(join(caseRoot, 'result.json'), snapshot.result)
        await writeSnapshot(join(caseRoot, 'guard.json'), {
          status: snapshot.result.status, billingStatus: snapshot.result.billingStatus,
          leaseStatus: snapshot.leaseStatus, usage: stateUsage(snapshot.state),
          estimatedCredits: weightedV1Credits(stateUsage(snapshot.state)), at: new Date().toISOString(),
        })
      },
      creditLimit: observedCaseGuardCredits, runTimeoutMs: maxWaitMs, drainTimeoutMs: 120_000,
    })
    terminal = observation.snapshot?.result
    assert(observation.snapshot, 'No lifecycle evidence')
    assert.deepEqual(observation.snapshot.result.handle, handle, 'Terminal handle mismatch')
    const verified = verifyEvaluationTerminal(start, observation.snapshot)
    await writeImmutable(join(caseRoot, 'terminal-verification.json'), verified)
    lifecycleVerified = true
  } catch {
    lifecycleError = 'lifecycle_finalization_unverified'
    // observeEvaluation already owns stop/drain. This is only for errors before observation.
    if (handle && !observation) {
      try { await stage.stop({ handle, requestId: `runner-stop-${start.requestId}`, reason: 'runner_error' }) } catch {}
    }
    await writeSnapshot(join(caseRoot, 'runner-error.json'), { code: lifecycleError })
  }
  const stopRequested = observation?.stopRequested ?? Boolean(lifecycleError)
  const stopReason = observation?.stopReason
  const stopError = observation?.stopFailed ? 'public_stop_failed' : undefined

  let state: any = { attempts: [] }
  let postflightError: string | undefined
  try { state = await readJson<any>(join(caseRoot, 'state.json')) }
  catch (error) { postflightError = error instanceof Error ? error.message : 'final state unavailable' }
  const configCheck = await verifyEvaluationConfiguration(before,
    async () => summarizeModelConfigs(await infrastructure.readModelConfigs()))
  // Persist unchanged / changed / unknown independently of report status.
  await writeImmutable(join(caseRoot, 'configuration-verification.json'), configCheck)
  if (configCheck.status !== 'unknown') await writeImmutable(join(caseRoot, 'model-config-after.json'), configCheck.after)
  else postflightError = configCheck.error
  const configUnchanged = configCheck.status === 'unknown' ? null : configCheck.status === 'unchanged'
  let creditsMicro = 0
  try { creditsMicro = receiptCreditsMicro(state) }
  catch (error) { postflightError ??= error instanceof Error ? error.message : 'receipt verification failed' }
  let evidence = { requestCount: 0, credentialLeak: true, rawFiles: 0, captureLimitHit: true }
  try { evidence = await scanEvidence(caseRoot, resolved.env) }
  catch (error) { postflightError ??= error instanceof Error ? error.message : 'evidence scan failed' }
  const terminalText = JSON.stringify({ code: terminal?.code, reason: terminal?.reason, status: terminal?.status }).toLowerCase()
  const runtimeGuardHit = evidence.captureLimitHit || /budget|timeout|turn_limit|request.?limit|messages request|protocol_error|usage_unavailable|model_error|process_exit|execution_failed|billing_failed/u.test(terminalText)
  try { await runFreeze() } catch { postflightError ??= 'frozen_source_drift' }
  const integrityOk = lifecycleVerified && !lifecycleError && !observation?.observationFailed && !observation?.drainTimedOut &&
    !postflightError && configUnchanged === true && !evidence.credentialLeak &&
    evidence.requestCount > 0 && evidence.requestCount <= 24 && !stopError && !stopRequested && !runtimeGuardHit
  const completed = {
    case: caseId, model, candidateId: candidate.id, candidatePromptSha256: candidate.candidatePromptSha256,
    status: terminal?.status ?? 'unknown', billingStatus: terminal?.billingStatus ?? 'unknown',
    unsettled: !lifecycleVerified, lifecycleVerified, lifecycleError, configStatus: configCheck.status, configUnchanged,
    creditsMicro: lifecycleVerified ? creditsMicro : null, knownReceiptedCreditsMicro: creditsMicro,
    requestCount: evidence.requestCount, rawFiles: evidence.rawFiles,
    credentialLeak: evidence.credentialLeak, captureLimitHit: evidence.captureLimitHit,
    runtimeGuardHit, integrityOk,
    ...(postflightError ? { postflightError } : {}),
    estimatedCredits: weightedV1Credits(stateUsage(state)), stopRequested, ...(stopReason ? { stopReason } : {}), ...(stopError ? { stopError } : {}),
    handle, at: new Date().toISOString(),
  }
  await writeImmutable(join(caseRoot, 'completed.json'), completed)
  console.log(JSON.stringify({ kind: 'completed', ...completed }))
  if (completed.unsettled) throw new Error('Review usage remains unsettled; no retry is permitted')
  if (!completed.integrityOk) throw new Error('Review evidence/configuration integrity check failed; no follow-up case is permitted')
}
