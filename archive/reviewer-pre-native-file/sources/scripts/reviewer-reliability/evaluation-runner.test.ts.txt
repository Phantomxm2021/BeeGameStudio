function authFixture() {
  let time = 1_000_000
  let records: EvaluationSession[] = [{ id: 'test_session', ownerId: 'test-owner', accessToken: 'old-fake-token',
    expiresAt: time + 60_000, sessionExpiresAt: time + 600_000 }]
  const requests: { url: string; method: string; redirect: unknown }[] = []
  let responseOwner = 'test-owner', failure = 0, persist = true
  const credential = createEvaluationCredential({ ownerId: 'test-owner', runtimeOrigin: 'http://127.0.0.1:62174',
    frontendOrigin: 'http://127.0.0.1:62173', now: () => time, loadSessions: async () => structuredClone(records),
    fetchImpl: (async (url, init) => {
      requests.push({ url: String(url), method: String(init?.method), redirect: init?.redirect })
      expect(new Headers(init?.headers).get('cookie')).toBe('beegame_session=test_session')
      expect(new Headers(init?.headers).get('origin')).toBe('http://127.0.0.1:62173')
      if (failure) return new Response('sensitive-upstream-body-fake-token', { status: failure })
      if (init?.method === 'POST' && persist) records[0] = { ...records[0]!, expiresAt: time + 90_000, accessToken: 'new-fake-token' }
      return Response.json({ authenticated: true, user: { id: responseOwner } })
    }) as typeof fetch })
  return { credential, requests, advance: (ms: number) => { time += ms },
    setRecords: (value: EvaluationSession[]) => { records = value }, getRecords: () => records,
    wrongOwner: () => { responseOwner = 'other-owner' }, fail: (status: number) => { failure = status },
    noPersist: () => { persist = false } }
}

describe('evaluation authentication: all tokens/fetches are synthetic', () => {
  test('fresh session needs no refresh; expired access token refreshes via the original local routes', async () => {
    const f = authFixture()
    expect(await f.credential.getValidAccessToken()).toBe('old-fake-token')
    expect(f.requests).toHaveLength(0)
    f.advance(70_000)
    expect(await f.credential.getValidAccessToken()).toBe('new-fake-token')
    expect(f.requests.map(value => [value.url, value.method, value.redirect])).toEqual([
      ['http://127.0.0.1:62174/api/auth/session', 'GET', 'error'],
      ['http://127.0.0.1:62174/api/auth/session/refresh', 'POST', 'error'],
    ])
  })
  test('refreshes within the 30-second safety margin', async () => {
    const f = authFixture(); f.advance(35_000)
    expect(await f.credential.getValidAccessToken()).toBe('new-fake-token')
  })
  test('forwards forceRefresh even when access token looks fresh, coalesces concurrent refreshes', async () => {
    const f = authFixture()
    expect(await Promise.all(Array.from({ length: 8 }, () => f.credential.getValidAccessToken({ forceRefresh: true }))))
      .toEqual(Array(8).fill('new-fake-token'))
    expect(f.requests).toHaveLength(2)
  })
  test('does not silently switch sessions after the pinned session disappears', async () => {
    const f = authFixture(); await f.credential.getValidAccessToken()
    f.setRecords([{ ...f.getRecords()[0]!, id: 'another_session' }])
    await expect(f.credential.getValidAccessToken()).rejects.toThrow('evaluation_auth_unavailable')
    expect(f.requests).toHaveLength(0)
  })
  test('expired login is not the same as expired access token', async () => {
    const f = authFixture(); f.advance(700_000)
    await expect(f.credential.getValidAccessToken()).rejects.toThrow('evaluation_auth_unavailable')
    expect(f.requests).toHaveLength(0)
  })
  test('owner mismatch stops before the refresh POST', async () => {
    const f = authFixture(); f.wrongOwner()
    await expect(f.credential.getValidAccessToken({ forceRefresh: true })).rejects.toThrow('evaluation_auth_unavailable')
    expect(f.requests).toHaveLength(1)
  })
  test.each([401, 403, 503])('HTTP %i failure fails closed without exposing response body', async status => {
    const f = authFixture(); f.fail(status)
    await expect(f.credential.getValidAccessToken({ forceRefresh: true })).rejects.toThrow(/^evaluation_auth_unavailable$/)
    expect(f.requests).toHaveLength(1)
  })
  test('successful refresh response without persisted fresh token is not success', async () => {
    const f = authFixture(); f.advance(70_000); f.noPersist()
    await expect(f.credential.getValidAccessToken()).rejects.toThrow('evaluation_auth_unavailable')
  })
  test('rejects nonlocal and credential-bearing origins before any fetch', () => {
    for (const runtimeOrigin of ['https://example.com', 'http://evil@127.0.0.1:62174', 'http://127.0.0.1:62174/path'])
      expect(() => createEvaluationCredential({ ownerId: 'owner', runtimeOrigin, frontendOrigin: 'http://127.0.0.1:62173',
        loadSessions: async () => [], fetchImpl: (() => { throw Error('must not fetch') }) as unknown as typeof fetch })).toThrow('evaluation_auth_origin_invalid')
  })
  test('read-only 401 retries once with forced refresh, never more', async () => {
    const f = authFixture(); const tokens: string[] = []
    const value = await authenticatedEvaluationRead(f.credential, async token => {
      tokens.push(token)
      if (tokens.length === 1) throw { status: 401 }
      return 'configuration'
    })
    expect(value).toBe('configuration'); expect(tokens).toEqual(['old-fake-token', 'new-fake-token'])
    let calls = 0
    await expect(authenticatedEvaluationRead(f.credential, async () => { calls++; throw { status: 401 } }))
      .rejects.toThrow('evaluation_read_unavailable')
    expect(calls).toBe(2)
  })
  test('403 / 503 read failures do not cause refresh or repeated queries', async () => {
    for (const status of [403, 503]) {
      const f = authFixture(); let calls = 0
      await expect(authenticatedEvaluationRead(f.credential, async () => { calls++; throw { status } })).rejects.toThrow('evaluation_read_unavailable')
      expect(calls).toBe(1); expect(f.requests).toHaveLength(0)
    }
  })
  test('actual dashboard-store 401 shape triggers exactly one authenticated read retry', async () => {
    const f = authFixture(); const tokens: string[] = []
    const store = new SupabaseDashboardStore({ url: 'https://synthetic.invalid', anonKey: 'fake-anon',
      fetchImpl: async (_url, init) => {
        tokens.push(new Headers(init?.headers).get('authorization')!)
        return tokens.length === 1 ? new Response('synthetic-body', { status: 401 }) : Response.json([])
      } })
    expect(await authenticatedEvaluationRead(f.credential, token => store.withAuthToken(token).listRlsVisiblePublicModelConfigs())).toEqual([])
    expect(tokens).toEqual(['Bearer old-fake-token', 'Bearer new-fake-token'])
    expect(f.requests).toHaveLength(2)
  })
  test('does not mistake a 503 body mentioning 401 for an authentication failure', async () => {
    const f = authFixture()
    await expect(authenticatedEvaluationRead(f.credential, async () => {
      throw Error('Supabase request failed: 503 Service Unavailable - upstream noted 401')
    })).rejects.toThrow('evaluation_read_unavailable')
    expect(f.requests).toHaveLength(0)
  })
  test('formal Hono session routes rotate and persist synthetic credentials; no listening socket or real auth', async () => {
    const root = await mkdtemp(join(tmpdir(), 'reviewer-auth-fixture-'))
    const storePath = join(root, 'sessions.json')
    const savedFlag = process.env.BEEGAME_HTTPONLY_SESSIONS, savedKey = process.env.BEEGAME_CONFIG_ENCRYPTION_KEY
    process.env.BEEGAME_HTTPONLY_SESSIONS = '1'
    process.env.BEEGAME_CONFIG_ENCRYPTION_KEY = Buffer.alloc(32, 7).toString('base64')
    let refreshes = 0
    try {
      const app = new Hono()
      registerBeeGameSessionRoutes(app, { sessionStorePath: storePath, supabaseUrl: 'https://synthetic.invalid', supabaseAnonKey: 'fake-anon',
        isOriginAllowed: origin => origin === 'http://127.0.0.1:62173',
        fetchImpl: async (url, init) => {
          if (String(url).endsWith('/auth/v1/user')) return Response.json({ id: 'test-owner', email: 'test@example.invalid' })
          expect(String(url)).toBe('https://synthetic.invalid/auth/v1/token?grant_type=refresh_token')
          expect(JSON.parse(String(init?.body)).refresh_token).toBe('synthetic-refresh')
          refreshes++
          return Response.json({ access_token: 'formal-new-token', refresh_token: 'synthetic-rotated-refresh', expires_in: 3600 })
        } })
      const login = await app.request('http://127.0.0.1:62174/api/auth/session', { method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ access_token: 'formal-old-token', refresh_token: 'synthetic-refresh', expires_in: 1 }) })
      expect(login.status).toBe(200)
      const credential = createEvaluationCredential({ ownerId: 'test-owner', runtimeOrigin: 'http://127.0.0.1:62174',
        frontendOrigin: 'http://127.0.0.1:62173',
        loadSessions: async () => Object.entries(JSON.parse(await readFile(storePath, 'utf8'))).map(([id, value]) => {
          const record = JSON.parse(decryptSecret(String(value), 'auth:session'))
          return { id, ownerId: record.user.id, accessToken: record.accessToken, expiresAt: record.expiresAt, sessionExpiresAt: record.sessionExpiresAt }
        }),
        fetchImpl: ((url, init) => app.request(new Request(String(url), init))) as typeof fetch })
      expect(await credential.getValidAccessToken()).toBe('formal-new-token')
      expect(await credential.getValidAccessToken()).toBe('formal-new-token')
      expect(refreshes).toBe(1)
      expect(await readFile(storePath, 'utf8')).not.toContain('formal-new-token')
    } finally {
      if (savedFlag === undefined) delete process.env.BEEGAME_HTTPONLY_SESSIONS; else process.env.BEEGAME_HTTPONLY_SESSIONS = savedFlag
      if (savedKey === undefined) delete process.env.BEEGAME_CONFIG_ENCRYPTION_KEY; else process.env.BEEGAME_CONFIG_ENCRYPTION_KEY = savedKey
      await rm(root, { recursive: true, force: true }) // Only this test's synthetic session store.
    }
  })
})

function terminalFixture() {
  const input = fixtureInputs().B24
  const start: SingleReviewStart = { protocolVersion: 1, ownerId: 'test-owner', projectId: 'test-project', requestId: 'offline-request',
    input, inputDigest: singleReviewInputDigest({ ownerId: 'test-owner', projectId: 'test-project', input }) }
  const report = { status: 'completed' as const, summary: 'Synthetic lifecycle fixture, not a semantic assessment',
    reviewedDocuments: input.documents.map(document => document.path), requirementsAssessment: 'reviewed' as const,
    findings: [], omissionCheck: 'Test only', limitations: ['Test only'] }
  const published = buildSingleReviewResult(start, report)
  const handle = { protocolVersion: 1 as const, producer: 'foundation-reviewer' as const, id: 'a2927bc5-c5e3-41d1-b3ae-aa5ba89d64c3' }
  const usage = { prompt_tokens: 100, completion_tokens: 10, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 110 }
  const attemptId = 'offline-attempt'
  const eventId = reviewDigest({ producer: 'foundation-reviewer', handle: handle.id, attemptId, usage })
  const receipt = { protocolVersion: 1, ownerId: start.ownerId, eventId, idempotencyKey: eventId, usage,
    pricingVersion: 'weighted-v1', usageSource: 'runtime_snapshot',
    recorded: { eventId: 'server-generated-id', creditsMicro: 15_000 }, debited: { eventId: 'server-generated-id', creditsMicro: 15_000 } }
  const snapshot: EvaluationSnapshot = { result: { status: 'completed', billingStatus: 'settled', handle, result: published }, leaseStatus: 'free',
    state: { attempts: [{ id: attemptId, usage, closed: { kind: 'closed', attemptId, usage, usageObserved: true,
      outcome: { kind: 'accepted', action: { action: 'final', report } } }, receipt, advanced: true }],
    transcript: [], stopped: false, report, published: published.resultId } }
  return { start, snapshot }
}

describe('finalization requires more than a terminal label', () => {
  test('accepts complete publication and bound metering, not semantic quality', () => {
    const f = terminalFixture()
    expect(verifyEvaluationTerminal(f.start, f.snapshot)).toMatchObject({ creditsMicro: 15_000, meteringVerified: true, semanticQuality: 'not_evaluated' })
  })
  test.each(['busy', 'stale'] as const)('rejects %s cleanup state', leaseStatus => {
    const f = terminalFixture(); f.snapshot.leaseStatus = leaseStatus
    expect(() => verifyEvaluationTerminal(f.start, f.snapshot)).toThrow('execution_cleanup_unverified')
  })
  test('rejects a settled label if metering was not observed or receipt is missing', () => {
    const f = terminalFixture(); f.snapshot.state.attempts[0]!.closed!.usageObserved = false
    expect(() => verifyEvaluationTerminal(f.start, f.snapshot)).toThrow('usage_not_sealed')
    f.snapshot.state.attempts[0]!.closed!.usageObserved = true; delete f.snapshot.state.attempts[0]!.receipt
    expect(() => verifyEvaluationTerminal(f.start, f.snapshot)).toThrow('receipt_missing')
  })
  test.each(['owner', 'event', 'usage', 'amount', 'debit', 'publication'] as const)('rejects %s mismatch', mutation => {
    const f = terminalFixture(), receipt = f.snapshot.state.attempts[0]!.receipt as any
    if (mutation === 'owner') receipt.ownerId = 'other'
    if (mutation === 'event') receipt.idempotencyKey = 'other'
    if (mutation === 'usage') receipt.usage = { ...receipt.usage, completion_tokens: 99 }
    if (mutation === 'amount') receipt.recorded.creditsMicro = 1
    if (mutation === 'debit') receipt.debited.eventId = 'other'
    if (mutation === 'publication') f.snapshot.state.published = 'different'
    expect(() => verifyEvaluationTerminal(f.start, f.snapshot)).toThrow()
  })
  test('settled semantic/format failure can be finalized without inventing a report', () => {
    const f = terminalFixture()
    f.snapshot.result = { status: 'needs_action', billingStatus: 'settled', handle: f.snapshot.result.handle, reason: 'invalid_output', resumable: false }
    delete f.snapshot.state.report; delete f.snapshot.state.published
    expect(verifyEvaluationTerminal(f.start, f.snapshot).creditsMicro).toBe(15_000)
  })
  test('duplicate attempts cannot be counted twice', () => {
    const f = terminalFixture(); f.snapshot.state.attempts.push(structuredClone(f.snapshot.state.attempts[0]!))
    expect(() => verifyEvaluationTerminal(f.start, f.snapshot)).toThrow('duplicate_attempt')
  })
  test('a zero-usage seal can use the formal skipped receipt', () => {
    const f = terminalFixture(), attempt = f.snapshot.state.attempts[0]!
    const usage = { prompt_tokens: 0, completion_tokens: 0, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 0 }
    attempt.usage = usage; attempt.closed!.usage = usage
    const eventId = reviewDigest({ producer: 'foundation-reviewer', handle: f.snapshot.result.handle.id, attemptId: attempt.id, usage })
    attempt.receipt = { protocolVersion: 1, ownerId: f.start.ownerId, eventId, idempotencyKey: eventId, usage,
      pricingVersion: 'weighted-v1', usageSource: 'runtime_snapshot', skipped: true, duplicate: false }
    expect(verifyEvaluationTerminal(f.start, f.snapshot).creditsMicro).toBe(0)
  })
  test('configuration unknown and changed remain distinct', async () => {
    expect((await verifyEvaluationConfiguration({ model: 'original' }, async () => ({ model: 'original' }))).status).toBe('unchanged')
    expect((await verifyEvaluationConfiguration({ model: 'original' }, async () => ({ model: 'different' }))).status).toBe('changed')
    expect(await verifyEvaluationConfiguration({}, async () => { throw Error('secret-token') })).toEqual({ status: 'unknown', error: 'configuration_unavailable' })
  })
  test('configuration timeout remains unknown, and late success cannot overwrite that result', async () => {
    let finish!: (value: object) => void
    const pending = new Promise<object>(resolve => { finish = resolve })
    const checked = await verifyEvaluationConfiguration({}, () => pending, 1)
    finish({})
    await Promise.resolve()
    expect(checked).toEqual({ status: 'unknown', error: 'configuration_unavailable' })
  })
})

describe('successor CLI has no implicit paid registration', () => {
  async function invoke(args: string[], batch = '') {
    const child = Bun.spawn([process.execPath, join(import.meta.dir, 'semantic-runner.ts'), ...args], {
      env: { ...process.env, BEEGAME_REVIEW_BATCH_ROOT: batch }, stdout: 'pipe', stderr: 'pipe',
    })
    return { code: await child.exited, stdout: await new Response(child.stdout).text(), stderr: await new Response(child.stderr).text() }
  }
  test('default preflight is zero-network and does not require real credentials', async () => {
    const result = await invoke(['--preflight'])
    expect(result.code).toBe(0)
    expect(JSON.parse(result.stdout)).toMatchObject({ inferenceCalls: 0, networkCalls: 0, status: 'offline_successor_ready_requires_new_registration' })
  })
  test('execute without a newly registered batch fails before infrastructure bootstrap', async () => {
    const result = await invoke(['--execute-once', 'case-1'])
    expect(result.code).toBe(1); expect(result.stderr).toContain('No new paid batch registered')
  })
  test('unapproved registration is rejected before any configuration or model access', async () => {
    const root = await mkdtemp(join(tmpdir(), 'reviewer-registration-fixture-'))
    try {
      await writeFile(join(root, 'registration.json'), JSON.stringify({ allowPaidExecution: false }))
      const result = await invoke(['--execute-once', 'case-1'], root)
      expect(result.code).toBe(1); expect(result.stderr).toContain('separately approved model/count/budget registration')
    } finally { await rm(root, { recursive: true, force: true }) }
  })
})

describe('observation and bounded stop/drain', () => {
  function harness() {
    let time = 0, reads = 0
    const stops: string[] = [], saved: EvaluationSnapshot[] = []
    const f = terminalFixture()
    const options = {
      read: async () => { reads++; return structuredClone(f.snapshot) },
      stop: async (reason: 'credit_guard' | 'timeout' | 'runner_error') => { stops.push(reason) },
      save: async (snapshot: EvaluationSnapshot) => { saved.push(snapshot) },
      now: () => time, sleep: async (ms: number) => { time += ms }, pollMs: 10,
      creditLimit: 12, runTimeoutMs: 20, drainTimeoutMs: 40,
    }
    return { options, f, stops, saved, time: () => time, reads: () => reads }
  }
  test('waits after completed until lease is free', async () => {
    const h = harness(); let n = 0
    h.options.read = async () => ({ ...h.f.snapshot, leaseStatus: ++n < 3 ? 'busy' : 'free' })
    const result = await observeEvaluation(h.options)
    expect(n).toBe(3); expect(h.stops).toEqual([]); expect(result.stopRequested).toBe(false)
  })
  test('waits for billing after non-running status', async () => {
    const h = harness(); let n = 0
    h.options.read = async () => {
      const snapshot = structuredClone(h.f.snapshot)
      if (++n < 3) { snapshot.result.billingStatus = 'pending'; delete snapshot.state.attempts[0]!.receipt }
      return snapshot
    }
    expect((await observeEvaluation(h.options)).snapshot?.result.billingStatus).toBe('settled')
    expect(n).toBe(3); expect(h.stops).toEqual([])
  })
  test('timeout requests stop once and gives settlement a new drain deadline', async () => {
    const h = harness()
    h.options.read = async () => {
      const snapshot = structuredClone(h.f.snapshot)
      if (h.time() < 50) snapshot.result = { status: 'running', billingStatus: 'pending', handle: snapshot.result.handle }
      return snapshot
    }
    const result = await observeEvaluation(h.options)
    expect(h.time()).toBe(50); expect(h.stops).toEqual(['timeout']); expect(result.drainTimedOut).toBe(false)
  })
  test('credit stop is once only, including tail-usage settlement', async () => {
    const h = harness(); h.options.creditLimit = 0.01
    h.options.read = async () => {
      const snapshot = structuredClone(h.f.snapshot)
      if (!h.stops.length) snapshot.result = { status: 'running', billingStatus: 'pending', handle: snapshot.result.handle }
      return snapshot
    }
    const result = await observeEvaluation(h.options)
    expect(h.stops).toEqual(['credit_guard']); expect(result.snapshot?.result.billingStatus).toBe('settled')
  })
  test('read exception stops and drains, but remains a failed observation', async () => {
    const h = harness(); let n = 0
    h.options.read = async () => { if (++n === 1) throw Error('synthetic read failure'); return h.f.snapshot }
    const result = await observeEvaluation(h.options)
    expect(h.stops).toEqual(['runner_error']); expect(result.observationFailed).toBe(true)
    expect(result.snapshot?.result.status).toBe('completed')
  })
  test('snapshot persistence failure triggers stop and preserves last in-memory evidence', async () => {
    const h = harness(); h.options.save = async () => { throw Error('synthetic disk failure') }
    const result = await observeEvaluation(h.options)
    expect(h.stops).toEqual(['runner_error']); expect(result.observationFailed).toBe(true)
    expect(result.drainTimedOut).toBe(true); expect(result.snapshot).toBeDefined()
  })
  test('stop failure is recorded and cannot cause a tight retry or unbounded wait', async () => {
    const h = harness()
    h.options.read = async () => { throw Error('synthetic read failure') }
    h.options.stop = async reason => { h.stops.push(reason); throw Error('synthetic stop failure') }
    const result = await observeEvaluation(h.options)
    expect(h.stops).toHaveLength(1); expect(result.stopFailed).toBe(true); expect(result.drainTimedOut).toBe(true)
    expect(h.time()).toBe(40)
  })
  test('settled label without sealed usage does not escape the drain bound', async () => {
    const h = harness(); h.f.snapshot.state.attempts[0]!.closed!.usageObserved = false
    const result = await observeEvaluation(h.options)
    expect(h.stops).toEqual(['timeout']); expect(result.drainTimedOut).toBe(true)
  })
})
