import { expect, test } from 'bun:test'
import { readFile } from 'node:fs/promises'
import { createSemanticPromptCandidate, SEMANTIC_PROMPT_EDITS } from './semantic-prompt-candidate'
import { screeningInput } from './fixtures'
import { createNativeToolReviewTask } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-tool-task'
import { captureSingleReviewInput, REVIEW_DOCUMENT_PATHS, reviewHash } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-contract'
import { presentReviewInput } from '../../packages/agent-workflow-server/src/beegame/stage-modules/foundation-reviewer/private/single-agent-input'

const boundary = '\nINPUT (each text field is one exact original line, excluding its newline; line is its one-based source position):\n'
function data(prompt: string) { return prompt.slice(prompt.indexOf(boundary)) }
function syntheticInput(content: string) {
  return captureSingleReviewInput({
    subject: { kind: 'initial', sourceResultId: 'candidate-offline', sourceDocumentRevision: reviewHash(content), confirmedBriefDigest: reviewHash('Synthetic test') },
    requirements: { status: 'unavailable', reason: 'Synthetic test' },
    documents: REVIEW_DOCUMENT_PATHS.map(path => ({ path, content })),
  })
}

test('exactly two additions; removing them recovers the B24 production prompt byte-for-byte', () => {
  const input = screeningInput(), baseline = createNativeToolReviewTask(input, [])
  const candidate = createSemanticPromptCandidate(input, [])
  let reversed = candidate.task.prompt
  for (const edit of SEMANTIC_PROMPT_EDITS) reversed = reversed.replace(`${edit.anchor} ${edit.addition}`, edit.anchor)
  expect(reversed).toBe(baseline.prompt)
  expect(data(candidate.task.prompt)).toBe(data(baseline.prompt))
  expect(candidate.baselinePromptSha256).not.toBe(candidate.candidatePromptSha256)
  expect(candidate.addedBytes).toBeGreaterThan(0)
  expect(candidate.addedBytes).toBeLessThan(600)
})

test('candidate preserves schema, tools descriptor and task keys without model-facing metadata', () => {
  const input = screeningInput(), baseline = createNativeToolReviewTask(input, [])
  const candidate = createSemanticPromptCandidate(input, []).task
  expect(Object.keys(candidate).sort()).toEqual(Object.keys(baseline).sort())
  expect(candidate.schema).toEqual(baseline.schema)
  expect(candidate.nativeTools).toEqual(baseline.nativeTools)
  expect(candidate.format).toBe(baseline.format)
  expect(candidate.prompt).not.toContain('semantic-instructions-r1')
})

test('candidate leaves caller input and continuing transcript unchanged', () => {
  const input = screeningInput()
  const transcript = [{ action: { action: 'tools' as const, rationale: 'Synthetic previous calculation',
    calls: [{ tool: 'calculate' as const, expression: { op: 'add' as const, args: [2, 3] } }] }, feedback: [{ result: { value: 5 } }] }]
  const before = structuredClone({ input, transcript })
  const baseline = createNativeToolReviewTask(input, transcript)
  const candidate = createSemanticPromptCandidate(input, transcript)
  expect({ input, transcript }).toEqual(before)
  expect(data(candidate.task.prompt)).toBe(data(baseline.prompt))
})

test('anchors inside untrusted source text cannot receive instruction edits', () => {
  const input = syntheticInput(SEMANTIC_PROMPT_EDITS.map(edit => edit.anchor).join('\n'))
  const baseline = createNativeToolReviewTask(input, [])
  const candidate = createSemanticPromptCandidate(input, [])
  expect(data(candidate.task.prompt)).toBe(data(baseline.prompt))
  expect(data(candidate.task.prompt)).not.toContain(SEMANTIC_PROMPT_EDITS[0].addition)
})

test('unavailable requirements and paged source policy stay intact', () => {
  const input = syntheticInput('Large synthetic paragraph\n'.repeat(3000))
  expect(presentReviewInput(input).paged).toBe(true)
  const baseline = createNativeToolReviewTask(input, [])
  const candidate = createSemanticPromptCandidate(input, [])
  expect(data(candidate.task.prompt)).toBe(data(baseline.prompt))
  expect(candidate.task.prompt).toContain('If original requirements are unavailable, disclose that limitation')
  expect(candidate.task.prompt).toContain('PAGED INPUT')
})

test('no new round, mandatory draft, numerical contract or budgets are introduced', () => {
  const candidate = createSemanticPromptCandidate(screeningInput(), []).task
  for (const statement of ['not a mandatory form, checklist or extra model round',
    'no draft or extra confirmation round is required', 'The audit time limit is 12 minutes.',
    'Only claim tool verification for successful calculations actually present in this transcript'])
    expect(candidate.prompt).toContain(statement)
})

test('sample answers are absent from the new instructions and the candidate module does not import fixtures', async () => {
  const additions = SEMANTIC_PROMPT_EDITS.map(edit => edit.addition).join('\n')
  expect(additions).not.toMatch(/S\d\d|B24|R41|R86|res_glow|maturity|planned|verified|UI_UX_SPEC|13|8.?10/)
  const source = await readFile(new URL('./semantic-prompt-candidate.ts', import.meta.url), 'utf8')
  expect(source).not.toMatch(/from\s+['"].*(?:fixtures|rubric|reproduce-semantic-report)/)
})

test('report validation and admission still have exactly the baseline behavior', () => {
  const input = screeningInput(), baseline = createNativeToolReviewTask(input, [])
  const candidate = createSemanticPromptCandidate(input, []).task
  const report = { status: 'completed', summary: 'Synthetic schema test, no semantic-quality claim',
    reviewedDocuments: [...REVIEW_DOCUMENT_PATHS], requirementsAssessment: 'reviewed', findings: [],
    omissionCheck: 'Synthetic test', limitations: ['Synthetic test'] }
  expect(candidate.admit(report)).toEqual(baseline.admit(report))
  expect(() => candidate.validateStructuredOutput({ action: 'final', report })).toThrow()
  const withFinding = { ...report, findings: [{ severity: 'HIGH', title: 'Synthetic finding', actual: 'a', expected: 'b', reasoning: 'c', impact: 'd',
    repair: { documentPath: 'docs/GDD.md', instruction: 'e' }, evidence: [{ path: 'docs/GDD.md', startLine: 1, endLine: 999 }] }] }
  expect(candidate.admit(withFinding)).toEqual(baseline.admit(withFinding))
  expect(candidate.admit(withFinding)).toMatchObject({ kind: 'rejected', diagnostic: { reason: 'citation_mismatch' } })
})
