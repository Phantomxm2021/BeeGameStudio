import { executeNativeReviewTool, REVIEW_TOOL_PROTOCOL } from '../single-agent-native-tools'
export const isNativeToolsFixture = () => process.env.BEEGAME_WORKFLOW_WORKER_TYPE === REVIEW_TOOL_PROTOCOL
export async function emitNativeToolsFixture({ input, report, transcript, mode, turns, usage, gddAudit = false }: any) {
  const descriptor = JSON.parse(process.env.BEEGAME_REVIEW_TOOL_SESSION!)
  const emit = (value: unknown) => process.stdout.write(JSON.stringify(value) + '\n')
  let callNumber = 0
  async function call(name: string, args: unknown) {
    const id = `tool-${turns}-${++callNumber}`
    emit({ type: 'assistant', message: { id: `request-${turns}`, content: [{ type: 'tool_use', id, name, input: args }] } })
    let result: any, error = false
    try { result = await executeNativeReviewTool(descriptor, name, args, id) }
    catch (failure) { result = String(failure); error = true }
    emit({ type: 'user', message: { role: 'user', content: [{ type: 'tool_result', tool_use_id: id,
      content: JSON.stringify(result), ...(error ? { is_error: true } : {}) }] } })
    return result
  }
  // The new wire report omits source quote bytes, which canonical admission copies.
  for (const finding of report.findings) for (const citation of finding.evidence) delete citation.quote
  if (mode !== 'missing-usage') emit({ type: 'assistant', message: { id: `request-${turns}`, usage, content: [] } })
  if (mode === 'normal' && !descriptor.recovery) await call('review_save_draft', report)
  if (mode === 'direct-final') report.summary = 'Reviewed 8 documents and bp_03'
  if (mode === 'invalid') report.reviewedDocuments = []
  if (mode === 'repeated-read') for (let i = 0; i < 3; i++)
    await call('review_read', { path: input.documents[0].path, startLine: 1, endLine: 1 })
  if (mode.startsWith('citation-') || mode === 'budget-preflight') {
    const rejected = transcript.some((entry: any) => entry.admitted === false)
    const bad = mode !== 'citation-correct' || !rejected
    if (mode === 'citation-final') await call('review_save_draft', report)
    if (rejected && !bad) await call('review_read', { path: input.documents[0].path, startLine: 1, endLine: 1 })
    report.findings = [{ severity: 'HIGH', title: 'Concrete gap', actual: 'Missing detail', expected: 'Specified detail',
      reasoning: 'Evidence supports the gap', impact: 'Cannot implement',
      evidence: [{ path: input.documents[0].path, startLine: bad ? 9999 : 1, endLine: bad ? 9999 : 1 }],
      repair: { documentPath: input.documents[0].path, instruction: 'Complete the detail' } }]
  }
  if (mode === 'numeric-correct') {
    const number = async (name: string, args: unknown) => (await call(name, args))[0].result.value as number
    const radiusSquared = await number('review_power', { left: 110, right: 2 })
    const coverage = async (distance: number) => {
      const distanceSquared = await number('review_power', { left: distance, right: 2 })
      const area = await number('review_subtract', { left: radiusSquared, right: distanceSquared })
      const half = await number('review_power', { left: area, right: 0.5 })
      return number('review_multiply', { values: [2, half] })
    }
    const left = await coverage(60), right = await coverage(70)
    const difference = await number('review_subtract', { left, right })
    report.findings = [{ severity: 'HIGH', title: 'Geometry gap', actual: 'Source geometry differs', expected: 'Consistent geometry',
      reasoning: `Coverage difference is ${difference.toFixed(2)}px`, impact: 'Coverage assumptions change',
      evidence: [{ path: input.documents[0].path, startLine: 1, endLine: 1 }],
      repair: { documentPath: input.documents[0].path, instruction: 'Align source geometry' } }]
    report.omissionCheck = `Rechecked source and ${difference.toFixed(2)}px difference`
  }
  if (mode === 'numeric-repeat') report.numericEvidence = { bindings: [], templates: {} }
  if (gddAudit) for (const finding of report.findings) {
    finding.documentPath = finding.repair.documentPath
    delete finding.repair
  }
  emit({ type: 'result', subtype: 'success', structured_output: report,
    ...(mode === 'missing-usage' ? {} : { modelUsage: { fixture: { input_tokens: usage.input_tokens * turns, output_tokens: usage.output_tokens * turns } } }) })
}
