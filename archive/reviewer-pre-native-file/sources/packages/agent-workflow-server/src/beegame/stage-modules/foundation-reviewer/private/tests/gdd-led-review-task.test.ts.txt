import { expect, test } from 'bun:test'
import { createGddLedReviewTask, gddLedReviewReportSchema, GDD_LED_REVIEW_POLICY } from '../gdd-led-review-task'
import { captureSingleReviewInput, REVIEW_DOCUMENT_PATHS, reviewHash } from '../single-agent-contract'
import { buildSingleReviewResult, singleReviewInputDigest, type SingleReviewStart } from '../single-agent-publication'

function fixture() {
  const input = captureSingleReviewInput({ subject: { kind: 'initial', sourceResultId: 'source',
    sourceDocumentRevision: 'a'.repeat(64), confirmedBriefDigest: reviewHash('Build a small game') },
    requirements: { status: 'available', content: 'Build a small game', digest: reviewHash('Build a small game') },
    documents: REVIEW_DOCUMENT_PATHS.map((path, index) => ({ path, content: `# Document ${index}\nOriginal rule ${index}` })) })
  const report = { status: 'completed' as const, summary: 'Reviewed the complete design', reviewedDocuments: [...REVIEW_DOCUMENT_PATHS],
    requirementsAssessment: 'reviewed' as const, findings: [{ severity: 'HIGH' as const, title: 'Specific defect',
      actual: 'Observed source behavior', expected: 'Supported requirement', reasoning: 'Reason based on original evidence', impact: 'Implementation consequence',
      evidence: [{ path: REVIEW_DOCUMENT_PATHS[0], startLine: 2, endLine: 2 }],
      documentPath: REVIEW_DOCUMENT_PATHS[0] }],
    omissionCheck: 'Revisited the overall game and specialist details', limitations: [] }
  return { input, report }
}

test('new task carries complete source and GDD-led reasoning without a coverage/proof output contract', () => {
  const { input, report } = fixture(), task = createGddLedReviewTask(input, [])
  expect(task.prompt).toContain('GDD as the entry point, not as unquestionable authority')
  expect(task.prompt).toContain('Do not review the GDD in isolation')
  expect(task.prompt).toContain('including details not referenced by the GDD')
  expect(task.prompt).toContain('Confirmed user requirements constrain the design')
  expect(task.prompt).toContain('reasonable refinements')
  expect(task.prompt).toContain('which rules or documents are affected')
  expect(task.prompt).toContain('Your responsibility ends at the audit findings')
  expect(task.prompt).toContain('Revision owns deciding and implementing changes')
  expect(task.prompt).toContain('do not choose a side or invent a requirement')
  expect(task.prompt).toContain('Do not turn actual/expected')
  expect(task.prompt).toContain('same object, scope, coordinate frame and conditions')
  expect(task.prompt).toContain('including the GDD; do not substitute your overall understanding')
  for (const document of input.documents) expect(task.prompt).toContain(document.content.split('\n')[1]!)
  expect(task.prompt).toContain('Build a small game')
  const schema = JSON.stringify(task.schema)
  expect(schema).toContain('The existing requirement or constraint and its source')
  expect(schema).toContain('Do not list edits, alternative fixes or a preferred outcome')
  for (const field of ['sourceChecks', 'auditPlanDigest', 'evidenceProof', 'mechanicalCheckId', 'numericBindings', 'numericEvidence', 'repair', 'instruction', 'remediation'])
    expect(schema).not.toContain(`"${field}"`)
  expect(() => task.validateStructuredOutput(report)).not.toThrow()
  expect(() => gddLedReviewReportSchema.parse({ ...report, sourceChecks: [] })).toThrow()
})

test('ordinary report publishes audit facts and exact source without revision instructions', () => {
  const { input, report } = fixture(), task = createGddLedReviewTask(input, [])
  const decision = task.admit(report)
  expect(decision.kind).toBe('accepted')
  if (decision.action.action !== 'final') throw new Error('Expected final report')
  const start: SingleReviewStart = { protocolVersion: 1, ownerId: 'owner', projectId: 'project', requestId: 'request', input,
    inputDigest: singleReviewInputDigest({ ownerId: 'owner', projectId: 'project', input }), validationPolicy: GDD_LED_REVIEW_POLICY }
  const result = buildSingleReviewResult(start, decision.action.report)
  expect(result.audit.policy).toBe(GDD_LED_REVIEW_POLICY)
  expect(result.audit.validation).toBeUndefined()
  expect(result.issues[0]!.issue).toMatchObject({ actual: report.findings[0]!.actual, expected: report.findings[0]!.expected,
    reasoning: report.findings[0]!.reasoning, impact: report.findings[0]!.impact,
    source: { documentPath: report.findings[0]!.documentPath }, evidence: [{ ...report.findings[0]!.evidence[0], quote: 'Original rule 0' }] })
  expect(result.issues[0]!.issue).not.toHaveProperty('remediation')
  expect(decision.action.report.findings[0]).not.toHaveProperty('repair')
  expect(result.documents.map(({ path, content }) => ({ path, content }))).toEqual(input.documents)
  const rejected = task.admit({ ...report, findings: [{ ...report.findings[0], evidence: [{ path: REVIEW_DOCUMENT_PATHS[0], startLine: 999, endLine: 999 }] }] })
  expect(rejected.kind).toBe('rejected')
  expect(() => task.admit({ ...report, status: 'blocked', findings: [], limitations: ['A design decision is missing'] })).toThrow()
  const continued = createGddLedReviewTask(input, [{ action: rejected.action, admitted: false, feedback: 'Correct the invalid reference' }])
  expect(continued.nativeTools.recovery).toEqual(task.nativeTools.recovery)
  expect(continued.prompt).toContain('Correct the invalid reference')
})

test('new output rejects repair plans rather than renaming or silently stripping them', () => {
  const { input, report } = fixture(), task = createGddLedReviewTask(input, [])
  for (const extension of [
    { repair: { documentPath: REVIEW_DOCUMENT_PATHS[0], instruction: 'Keep coordinates and lower the constraint to 60' } },
    { instruction: 'Choose Esc cancellation before pause' },
    { remediation: { kind: 'patch', documentPath: REVIEW_DOCUMENT_PATHS[0] } },
  ]) {
    const value = { ...report, findings: [{ ...report.findings[0], ...extension }] }
    expect(() => task.validateStructuredOutput(value)).toThrow()
    expect(() => task.admit(value)).toThrow()
  }
})
