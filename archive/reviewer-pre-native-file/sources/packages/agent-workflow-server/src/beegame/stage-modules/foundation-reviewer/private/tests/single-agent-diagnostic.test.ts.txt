import { expect, test } from 'bun:test'
import { z } from 'zod/v4'
import { diagnoseReviewOutput, ReviewAdmissionError, reviewOutputDiagnosticSchema, reviewNativeSummarySchema } from '../single-agent-diagnostic'

test('native summary allows only bounded classifications and counts', () => {
  const summary = { modelMessages: 2, structuredOutputCalls: 1, structuredOutputSuccesses: 1,
    structuredOutputFailures: 0, lastStopReason: 'tool_use', usageSource: 'message_snapshots' }
  expect(reviewNativeSummarySchema.safeParse(summary).success).toBe(true)
  for (const value of [{ ...summary, text: 'PRIVATE' }, { ...summary, lastStopReason: 'PRIVATE' }, { ...summary, modelMessages: -1 }])
    expect(reviewNativeSummarySchema.safeParse(value).success).toBe(false)
})
import { singleReviewEventSchema } from '../single-agent-journal'
import { Ajv } from 'ajv'
import { focusSchemaErrors } from '../../../../../../../builtin-tools/src/tools/SyntheticOutputTool/schemaDiagnostics'
import { createSingleReviewTask } from '../single-agent-task'
import { REVIEW_DOCUMENT_PATHS, type SingleReviewInput } from '../single-agent-contract'

test('native StructuredOutput Ajv can register the exact generated action schema', () => {
  const input: SingleReviewInput = { subject: { kind: 'initial', sourceResultId: 'source', sourceDocumentRevision: 'a'.repeat(64), confirmedBriefDigest: 'b'.repeat(64) },
    requirements: { status: 'unavailable', reason: 'Offline schema test' }, documents: REVIEW_DOCUMENT_PATHS.map(path => ({ path, content: '# Document' })) }
  const task = createSingleReviewTask(input, [])
  const ajv = new Ajv({ allErrors: true })
  expect(ajv.validateSchema(task.schema)).toBe(true)
  const validate = ajv.compile(task.schema)
  expect(validate({ action: 'tools', rationale: 'Check original evidence', calls: [{ tool: 'read', path: 'docs/GDD.md', startLine: 1, endLine: 1 }] })).toBe(true)
  expect(validate({ action: 'final', report: {} })).toBe(false)
  const reference = { path: 'docs/GDD.md', startLine: 1, endLine: 1 }
  const amendment = { action: 'amend_draft', draftDigest: 'a'.repeat(64),
    citations: [{ findingIndex: 0, citationIndex: 0, reference }] }
  expect(validate(amendment)).toBe(true)
  expect(validate({ ...amendment, citations: [{ ...amendment.citations[0], reference: { ...reference, quote: 'Model must not copy bytes' } }] })).toBe(false)
  expect(validate({ action: 'confirm_draft', draftDigest: 'a'.repeat(64), omissionCheck: 'Checked omissions and false positives' })).toBe(true)
  const tools = { action: 'tools', rationale: 'Verify expression', calls: [{ tool: 'calculate', expression: { op: 'multiply', args: [6, 8] } }] }
  const extra = { ...tools, extra: true }
  expect(validate(extra)).toBe(false)
  expect(validate.errors!.length).toBeGreaterThan(20)
  const focused = focusSchemaErrors(task.schema, extra, validate.errors!)
  expect(focused).toHaveLength(1)
  expect(focused[0]!.keyword).toBe('additionalProperties')
  const wrong = { ...tools, calls: [{ tool: 'calculate', expression: { op: 'multiply', args: ['6', 8] } }] }
  expect(validate(wrong)).toBe(false)
  const leafErrors = focusSchemaErrors(task.schema, wrong, validate.errors!)
  expect(leafErrors.length).toBeLessThan(5)
  expect(leafErrors.every(error => error.instancePath === '/calls/0/expression/args/0')).toBe(true)
  // Diagnostic filtering never normalizes or repairs the original payload.
  expect(validate(wrong)).toBe(false)
  expect(wrong.calls[0]!.expression.args[0]).toBe('6')
  const nested = { ...tools, calls: [{ tool: 'calculate', expression: { op: 'multiply', args: [
    { op: 'subtract', args: ['6', 8] }, 2,
  ] } }] }
  expect(validate(nested)).toBe(false)
  const nestedErrors = focusSchemaErrors(task.schema, nested, validate.errors!)
  expect(nestedErrors.length).toBeGreaterThan(0)
  expect(nestedErrors.every(error => error.instancePath === '/calls/0/expression/args/0/args/0')).toBe(true)
  const wrongArity = { ...tools, calls: [{ tool: 'calculate', expression: { op: 'multiply', args: [
    { op: 'subtract', args: [6, 8, 2] }, 2,
  ] } }] }
  expect(validate(wrongArity)).toBe(false)
  const arityErrors = focusSchemaErrors(task.schema, wrongArity, validate.errors!)
  expect(arityErrors).toHaveLength(1)
  expect(arityErrors[0]).toMatchObject({ instancePath: '/calls/0/expression/args/0/args', keyword: 'maxItems' })
})

test('diagnostics retain precise safe admission locations, not exception or model text', () => {
  expect(diagnoseReviewOutput(new ReviewAdmissionError({ stage: 'admission', reason: 'citation_mismatch', findingIndex: 2, citationIndex: 1 })))
    .toEqual({ stage: 'admission', reason: 'citation_mismatch', findingIndex: 2, citationIndex: 1 })
  expect(diagnoseReviewOutput(new Error('PRIVATE_PROVIDER_SECRET'))).toEqual({ stage: 'admission', reason: 'unknown' })
  const result = z.object({ action: z.literal('draft') }).strict().safeParse({ action: 'PRIVATE_VALUE', PRIVATE_FIELD: 'secret' })
  if (result.success) throw new Error('Expected schema failure')
  const diagnostic = diagnoseReviewOutput(result.error)
  expect(diagnostic.stage).toBe('payload_schema')
  expect(diagnostic.schemaCodes).toContain('invalid_value')
  expect(JSON.stringify(diagnostic)).not.toContain('PRIVATE')
  expect(reviewOutputDiagnosticSchema.safeParse({ ...diagnostic, message: 'raw provider output' }).success).toBe(false)
})

test('sealed failure retains bounded diagnostics while original failure records stay readable', () => {
  const event = { kind: 'closed', attemptId: 'attempt', usageObserved: true,
    usage: { prompt_tokens: 1, completion_tokens: 1, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 2 },
    outcome: { kind: 'failed', code: 'invalid_output' } }
  expect(singleReviewEventSchema.safeParse(event).success).toBe(true)
  expect(singleReviewEventSchema.parse({ ...event, outcome: { ...event.outcome, diagnostic: { stage: 'missing_structured_output' } } }))
    .toMatchObject({ outcome: { diagnostic: { stage: 'missing_structured_output' } } })
  expect(singleReviewEventSchema.safeParse({ ...event, outcome: { ...event.outcome, diagnostic: { stage: 'admission', reason: 'arbitrary secret' } } }).success).toBe(false)
})
