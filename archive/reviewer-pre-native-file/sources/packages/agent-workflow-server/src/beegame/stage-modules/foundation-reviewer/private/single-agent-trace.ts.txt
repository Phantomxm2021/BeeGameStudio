import { constants, openSync, writeSync, closeSync } from 'node:fs'
import { randomUUID } from 'node:crypto'
import { join } from 'node:path'
import { StringDecoder } from 'node:string_decoder'
import { NATIVE_DIAGNOSTIC_PREFIX, parseNativeDiagnostic, nativeErrorCategory,
  type NativeDiagnostic } from './single-agent-trace-protocol'

/** Separate from admission/billing evidence. No raw stderr is written to disk. */
export function createReviewTrace(directory: string) {
  const path = join(directory, `diagnostics-${randomUUID()}.jsonl`)
  const fd = openSync(path, constants.O_WRONLY | constants.O_CREAT | constants.O_EXCL | constants.O_NOFOLLOW, 0o600)
  const started = performance.now()
  let failed = false, closed = false, firstStdout = true
  let pending = '', truncated = false, bytes = 0, lines = 0
  const categories = new Map<NonNullable<NativeDiagnostic['category']>, number>()
  const decoder = new StringDecoder('utf8')
  function record(value: NativeDiagnostic) {
    if (closed || failed) return
    const safe = parseNativeDiagnostic(value)
    if (!safe) return
    try { writeSync(fd, JSON.stringify({ at: new Date().toISOString(), elapsedMs: Math.ceil(performance.now() - started), ...safe }) + '\n') }
    catch { failed = true }
  }
  function stderrLine(line: string) {
    if (!truncated && line.startsWith(NATIVE_DIAGNOSTIC_PREFIX)) {
      let event: NativeDiagnostic | undefined
      try { event = parseNativeDiagnostic(JSON.parse(line.slice(NATIVE_DIAGNOSTIC_PREFIX.length))) } catch { /* Metadata parse failure only. */ }
      record(event ?? { event: 'diagnostic_invalid' })
    } else {
      lines++
      const category = nativeErrorCategory(line)
      // Record the first occurrence only; final counts prove stderr was seen.
      if (!categories.has(category)) record({ event: 'stderr_summary', category, lines: 1 })
      categories.set(category, (categories.get(category) ?? 0) + 1)
    }
    truncated = false
  }
  function consume(chunk: string) {
    for (const fragment of chunk.split(/(?<=\n)/u)) {
      if (!truncated) {
        pending += fragment
        if (pending.length > 8192) { pending = pending.slice(0, 8192); truncated = true }
      }
      if (fragment.endsWith('\n')) { stderrLine(pending.trimEnd()); pending = '' }
    }
  }
  return {
    path, record,
    get failed() { return failed },
    stdout(chunk: Buffer) { if (firstStdout) { firstStdout = false; record({ event: 'stdout_first', bytes: chunk.length }) } },
    stderr(chunk: Buffer) { bytes += chunk.length; consume(decoder.write(chunk)) },
    close() {
      if (closed) return
      consume(decoder.end())
      if (pending || truncated) stderrLine(pending)
      record({ event: 'stderr_summary', bytes, lines })
      closed = true
      closeSync(fd)
    },
  }
}
