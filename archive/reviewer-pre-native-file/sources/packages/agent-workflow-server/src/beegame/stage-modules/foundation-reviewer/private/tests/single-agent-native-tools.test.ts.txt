const source = () => captureSingleReviewInput({ subject: { kind: 'initial', sourceResultId: 'native-tools-test',
  sourceDocumentRevision: 'a'.repeat(64), confirmedBriefDigest: 'b'.repeat(64) },
  requirements: { status: 'unavailable', reason: 'Test fixture' },
  documents: REVIEW_DOCUMENT_PATHS.map(path => ({ path, content: '# Source\nExact source evidence' })) })
const report = () => ({ status: 'completed', summary: 'Reviewed complete input', reviewedDocuments: [...REVIEW_DOCUMENT_PATHS],
  requirementsAssessment: 'unavailable', findings: [{ severity: 'HIGH', title: 'Evidence-bound finding', actual: 'Observed',
    expected: 'Required', reasoning: 'Compared original sources', impact: 'Delivery risk',
    evidence: [{ path: 'docs/GDD.md', startLine: 2, endLine: 2 }],
    repair: { documentPath: 'docs/GDD.md', instruction: 'Address documented mismatch' } }],
  omissionCheck: 'Revisited complete game', limitations: ['Requirements unavailable'] })
async function fixture(run: (descriptor: Awaited<ReturnType<typeof prepareReviewToolSession>>, root: string) => Promise<void>) {
  const root = await realpath(await mkdtemp(join(tmpdir(), 'review-native-tools-')))
  try { await run(await prepareReviewToolSession(root, 'test-session', source(), []), root) }
  finally { await rm(root, { recursive: true, force: true }) }
}
describe('native reviewer tools v2', () => {
  test('all model schemas are ordinary objects without unions, recursion or envelopes', () => {
    const schemas = [...reviewNativeToolSchemas().map(tool => tool.input_schema), createNativeToolReviewTask(source(), []).schema]
    for (const schema of schemas) {
      expect((schema as any).type).toBe('object')
      expect(JSON.stringify(schema)).not.toMatch(/"(?:oneOf|anyOf|\$ref|payload_json|numericEvidence|numericBindings)"/)
      expect(new Ajv({ allErrors: true }).validateSchema(schema)).toBe(true)
    }
  })
  test('new protocol explicitly uses named tools and final report only; old wrapper is rejected', () => {
    const task = createNativeToolReviewTask(source(), [])
    expect(task.prompt).toContain('review_subtract')
    expect(task.prompt).not.toContain('Available actions:')
    expect(task.admit(report()).kind).toBe('accepted')
    expect(() => task.admit({ action: 'final', report: report() })).toThrow()
    expect(() => task.admit({ ...report(), status: 'blocked' })).toThrow()
  })
  test('CLI tools are absent outside reviewer scope and invalid capability fails closed', () => {
    expect(createNativeReviewTools(value => value, {})).toEqual([])
    expect(createNativeReviewTools(value => value, { BEEGAME_REVIEW_TOOL_SESSION: '{}' })).toEqual([])
    expect(() => createNativeReviewTools(value => value, { BEEGAME_WORKFLOW_WORKER_TYPE: REVIEW_TOOL_PROTOCOL,
      BEEGAME_REVIEW_TOOL_SESSION: '{}' })).toThrow()
  })
  test('persist before return, deduplicate same ID, and reject changed-ID meaning', () => fixture(async (descriptor, root) => {
    const args = { left: 18, right: 11 }
    const result: any = await executeNativeReviewTool(descriptor, 'review_subtract', args, 'call-1')
    expect(result[0].result.value).toBe(7)
    expect((await readReviewToolRecords(descriptor))).toHaveLength(1)
    expect(await executeNativeReviewTool(descriptor, 'review_subtract', args, 'call-1')).toEqual(result)
    expect(await readdir(join(root, 'calls'))).toHaveLength(1)
    await expect(executeNativeReviewTool(descriptor, 'review_subtract', { left: 19, right: 11 }, 'call-1')).rejects.toThrow('reused')
  }))
  test('parallel invocations serialize durable sequence; results remain independently bound', () => fixture(async descriptor => {
    const values = await Promise.all([1, 2, 3].map(left => executeNativeReviewTool(descriptor, 'review_add', { values: [left, 10] }, 'call-'+left)))
    expect(values.map((value: any) => value[0].result.value)).toEqual([11,12,13])
    expect((await readReviewToolRecords(descriptor)).map(record => record.sequence)).toEqual([1,2,3])
  }))
  test('bounds paths and types, never coerces provider errors, and persists genuine execution failures', () => fixture(async descriptor => {
    for (const [name, args] of [
      ['review_read', { path: '/etc/passwd', startLine: 1, endLine: 1 }],
      ['review_add', { values: { item: ['18', '11'] } }],
      ['review_subtract', { left: '18', right: 11 }],
      ['review_subtract', { left: 18, right: 11, extra: 3 }],
      ['Bash', { command: 'echo unsafe' }],
    ] as const) await expect(executeNativeReviewTool(descriptor, name, args, 'invalid')).rejects.toThrow()
    expect(await readReviewToolRecords(descriptor)).toHaveLength(0)
    const result: any = await executeNativeReviewTool(descriptor, 'review_divide', { left: 1, right: 0 }, 'division-zero')
    expect(result[0].result).toBeUndefined(); expect(result[0].error).toContain('finite')
    const read: any = await executeNativeReviewTool(descriptor, 'review_read', { path: 'docs/GDD.md', startLine: 2, endLine: 2 }, 'read')
    expect(read[0].result.content).toBe('Exact source evidence')
  }))
  test('draft is bound state, not final completion; bad citations stay rejected', () => fixture(async descriptor => {
    const draft: any = await executeNativeReviewTool(descriptor, 'review_save_draft', report(), 'draft')
    expect(draft.admission).toBe('accepted'); expect(draft.draftDigest).toHaveLength(64)
    const invalid = report(); invalid.findings[0]!.evidence[0]!.endLine = 100
    const rejected: any = await executeNativeReviewTool(descriptor, 'review_save_draft', invalid, 'rejected')
    expect(rejected.admission).toBe('rejected')
    const records = await readReviewToolRecords(descriptor)
    expect(records.map(record => record.entry.action.action)).toEqual(['draft','draft'])
    expect(records[1]!.entry.admitted).toBe(false)
  }))
  test('tampered source, log result, and symlink records fail closed', () => fixture(async (descriptor, root) => {
    await executeNativeReviewTool(descriptor, 'review_add', { values: [18,11] }, 'call')
    const path = join(root, 'calls', '000001.json')
    const original = await readFile(path, 'utf8')
    const value = JSON.parse(original); value.entry.feedback[0].result.value = 999
    await writeFile(path, JSON.stringify(value))
    await expect(readReviewToolRecords(descriptor)).rejects.toThrow()
    await writeFile(path, original)
    await rm(path); await symlink(join(root, 'context.json'), path)
    await expect(readReviewToolRecords(descriptor)).rejects.toThrow()
  }))
  test('repeated calls persist a stop that cannot be bypassed by a final draft or a new session', () => fixture(async (descriptor, root) => {
    const args = { path: 'docs/GDD.md', startLine: 1, endLine: 1 }
    await executeNativeReviewTool(descriptor, 'review_read', args, 'one')
    await executeNativeReviewTool(descriptor, 'review_read', args, 'two')
    await expect(executeNativeReviewTool(descriptor, 'review_read', args, 'three')).rejects.toThrow('audit_progress_stalled')
    expect(await readReviewToolStop(descriptor)).toBe('audit_progress_stalled')
    await expect(executeNativeReviewTool(descriptor, 'review_save_draft', report(), 'bypass')).rejects.toThrow('audit_progress_stalled')
    const records = await readReviewToolRecords(descriptor)
    expect(records).toHaveLength(2)
    const resumed = await prepareReviewToolSession(join(root, 'resumed'), 'resumed', source(), records.map(record => record.entry))
    await expect(executeNativeReviewTool(resumed, 'review_read', args, 'fresh-id')).rejects.toThrow('audit_progress_stalled')
  }))
  test('no implicit cumulative call budget; aborted tool context still cannot execute', () => fixture(async (descriptor, root) => {
    await executeNativeReviewTool(descriptor, 'review_add', { values: [1, 2] }, 'one')
    const entry = (await readReviewToolRecords(descriptor))[0]!.entry
    const capped = await prepareReviewToolSession(join(root, 'capped'), 'capped', source(), Array(144).fill(entry))
    await executeNativeReviewTool(capped, 'review_add', { values: [3, 4] }, 'new')
    expect(await readReviewToolRecords(capped)).toHaveLength(1)
    const abortController = new AbortController(); abortController.abort()
    const tools = createNativeReviewTools(value => value, { BEEGAME_WORKFLOW_WORKER_TYPE: REVIEW_TOOL_PROTOCOL,
      BEEGAME_REVIEW_TOOL_SESSION: JSON.stringify(descriptor) })
    await expect(tools[0].call({ path: 'docs/GDD.md', startLine: 1, endLine: 1 }, { toolUseId: 'aborted', abortController })).rejects.toThrow('cancelled')
    expect(await readReviewToolRecords(descriptor)).toHaveLength(1)
  }))
  test('recovery mode is opt-in, exposes verified receipts, and makes context reads idempotent', async () => {
    const root = await realpath(await mkdtemp(join(tmpdir(), 'review-native-recovery-')))
    try {
      const descriptor = await prepareReviewToolSession(root, 'recovery-session', source(), [], {
        workId: 'recoverable-work', prompt: 'Review the assigned batch and submit its bounded result.',
      })
      expect(REVIEW_NATIVE_RECOVERY_TOOL_NAMES).toContain('review_recover_context')
      expect(REVIEW_NATIVE_RECOVERY_TOOL_NAMES).not.toContain('review_save_draft')
      expect(await readReviewRecoveryContext(descriptor)).toEqual({
        workId: 'recoverable-work', prompt: 'Review the assigned batch and submit its bounded result.', calculations: [],
      })
      const first: any = await executeNativeReviewTool(descriptor, 'review_recover_context', {}, 'recover-1')
      const retry: any = await executeNativeReviewTool(descriptor, 'review_recover_context', {}, 'recover-1')
      expect(retry).toEqual(first)
      expect(await readdir(join(root, 'calls'))).toHaveLength(0)
      expect(await readdir(join(root, 'progress'))).toHaveLength(1)
      await expect(executeNativeReviewTool(descriptor, 'review_save_draft', report(), 'draft')).rejects.toThrow('allowlist')

      const firstCalculation: any = await executeNativeReviewTool(descriptor, 'review_add', { values: [18, 11] }, 'math-1')
      expect(firstCalculation[0].receipt).toBe(1)
      expect(Number.isInteger(firstCalculation[0].receipt)).toBe(true)
      const cachedCalculation: any = await executeNativeReviewTool(descriptor, 'review_add', { values: [18, 11] }, 'math-2')
      expect(cachedCalculation[0].result.value).toBe(29)
      expect(cachedCalculation[0].receipt).toBe(1)
      expect(cachedCalculation[0].cached).toBe(true)
      const context: any = await executeNativeReviewTool(descriptor, 'review_recover_context', {}, 'recover-2')
      expect(context.calculations).toEqual([expect.objectContaining({ receipt: 1, value: 29 })])
    } finally { await rm(root, { recursive: true, force: true }) }
  })
  test('recovery progress warns at twelve distinct retries and stops at thirteen', async () => {
    const root = await realpath(await mkdtemp(join(tmpdir(), 'review-native-recovery-guard-')))
    try {
      const descriptor = await prepareReviewToolSession(root, 'recovery-guard', source(), [], {
        workId: 'recoverable-guard-work', prompt: 'Review this batch.',
      })
      for (let index = 1; index <= 12; index++) {
        const result: any = await executeNativeReviewTool(descriptor, 'review_recover_context', {}, `recover-${index}`)
        if (index < 12) expect(result.warning).toBeUndefined()
        else expect(result.warning).toContain('no new information')
      }
      await expect(executeNativeReviewTool(descriptor, 'review_recover_context', {}, 'recover-13')).rejects.toThrow('audit_progress_stalled')
      expect(await readReviewToolStop(descriptor)).toBe('audit_progress_stalled')
      expect(await readdir(join(root, 'progress'))).toHaveLength(13)
    } finally { await rm(root, { recursive: true, force: true }) }
  })
  test('recovery progress also stops identical bounded failures', async () => {
    const root = await realpath(await mkdtemp(join(tmpdir(), 'review-native-recovery-failure-')))
    try {
      const descriptor = await prepareReviewToolSession(root, 'recovery-failure', source(), [], {
        workId: 'recoverable-failure-work', prompt: 'Review this batch.',
      })
      for (let index = 1; index <= 12; index++) {
        const result: any = await executeNativeReviewTool(descriptor, 'review_divide', { left: 1, right: 0 }, `failure-${index}`)
        if (index < 12) expect(result[0].warning).toBeUndefined()
        else expect(result[0].warning).toContain('no new information')
      }
      await expect(executeNativeReviewTool(descriptor, 'review_divide', { left: 1, right: 0 }, 'failure-13')).rejects.toThrow('audit_progress_stalled')
      expect(await readReviewToolStop(descriptor)).toBe('audit_progress_stalled')
    } finally { await rm(root, { recursive: true, force: true }) }
  })
  test('recovery progress counts alternating cached calculations as repeated no-progress', async () => {
    const root = await realpath(await mkdtemp(join(tmpdir(), 'review-native-recovery-alternating-')))
    try {
      const descriptor = await prepareReviewToolSession(root, 'recovery-alternating', source(), [], {
        workId: 'recoverable-alternating-work', prompt: 'Review this batch.',
      })
      const calculations = [{ values: [18, 11] }, { values: [20, 2] }]
      await executeNativeReviewTool(descriptor, 'review_add', calculations[0], 'seed-a')
      await executeNativeReviewTool(descriptor, 'review_add', calculations[1], 'seed-b')
      for (let index = 0; index < 11; index++) {
        const result: any = await executeNativeReviewTool(descriptor, 'review_add', calculations[index % 2]!, `cached-${index}`)
        expect(result[0].cached).toBe(true)
        if (index < 10) expect(result[0].warning).toBeUndefined()
        else expect(result[0].warning).toContain('no new information')
      }
      await expect(executeNativeReviewTool(descriptor, 'review_add', calculations[1], 'cached-11')).rejects.toThrow('audit_progress_stalled')
      expect(await readReviewToolStop(descriptor)).toBe('audit_progress_stalled')
      expect(await readdir(join(root, 'progress'))).toHaveLength(14)
    } finally { await rm(root, { recursive: true, force: true }) }
  })
  test('old invalid binary arity remains readable and replayable, but cannot be newly admitted or executed', () => fixture(async (_descriptor, root) => {
    const input = source()
    const start = { protocolVersion: 1 as const, ownerId: 'owner', projectId: 'project', requestId: 'legacy', input,
      inputDigest: singleReviewInputDigest({ ownerId: 'owner', projectId: 'project', input }) }
    const journal = await openSingleReviewJournal(root, 'legacy', start)
    const action = { action: 'tools' as const, rationale: 'Historical schema allowed three operands',
      calls: [{ tool: 'calculate' as const, expression: { op: 'subtract' as const, args: [520, 40, 10] } }] }
    const usage = { prompt_tokens: 20, completion_tokens: 10, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 30 }
    await journal.append('claim', { kind: 'claim', attemptId: 'old' })
    await journal.append('closed', { kind: 'closed', attemptId: 'old', usage, usageObserved: true, outcome: { kind: 'accepted', action: action as any } })
    await journal.append('settled', { kind: 'settled', attemptId: 'old', receipt: {} })
    const feedback = [{ call: action.calls[0], error: 'This operator requires exactly two operands' }]
    await journal.append('advance', { kind: 'advance', attemptId: 'old', feedback })
    const state = await (await openSingleReviewJournal(root, 'legacy', start, true)).read()
    expect(state.transcript[0]).toEqual({ action, feedback })
    expect(createContinuingSingleReviewTask(input, []).replay(action)).toEqual({ kind: 'accepted', action })
    expect(() => createContinuingSingleReviewTask(input, []).admit(action)).toThrow()
    expect(() => executeSingleReviewTool(input, action.calls[0])).toThrow()
  }))
})

const validateCalculation = new Ajv({ allErrors: true }).compile(z.toJSONSchema(reviewToolSchema, { target: 'draft-7' }))
for (const op of ['add','subtract','multiply','divide','min','max','power']) for (const count of [0,1,2,3,32,33]) {
  test(`calculation schema agrees with executor: ${op}, ${count} operands`, () => {
    const call = { tool: 'calculate', expression: { op, args: Array(count).fill(1) } }
    let executes = true
    try { executeSingleReviewTool(source(), call) } catch { executes = false }
    expect(validateCalculation(call)).toBe(executes)
    expect(executes).toBe(['subtract','divide','power'].includes(op) ? count === 2 : count >= 2 && count <= 32)
  })
}
for (const op of ['subtract','divide','power']) for (const count of [2,3]) test(`nested binary ${op}/${count}`, () => {
  const call = { tool: 'calculate', expression: { op: 'add', args: [1, { op, args: Array(count).fill(1) }] } }
  expect(validateCalculation(call)).toBe(count === 2)
  if (count === 2) expect(() => executeSingleReviewTool(source(), call)).not.toThrow()
  else expect(() => executeSingleReviewTool(source(), call)).toThrow()
})
