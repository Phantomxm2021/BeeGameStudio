import { expect, test } from 'bun:test'
import { consumedReviewTime, consumedReviewTokens, reviewProgressStalled, nextReviewTokenReserve, reviewCompletionForecast } from '../single-agent-budget'
import type { SingleReviewAttempt } from '../single-agent-journal'
import type { SingleReviewAction } from '../single-agent-report'

test('active execution time and usage add across sealed attempts, not just the latest turn', () => {
  const attempts: SingleReviewAttempt[] = [1, 2, 3].map(index => ({ id: String(index), advanced: false, closed: {
    kind: 'closed', attemptId: String(index), elapsedMs: 240_000, usageObserved: true,
    usage: { prompt_tokens: 10, completion_tokens: 5, cache_read_tokens: 2, cache_creation_tokens: 1, total_tokens: 18 },
    outcome: { kind: 'failed', code: 'stopped' },
  } }))
  expect(consumedReviewTime({ attempts })).toBe(720_000)
  expect(consumedReviewTokens({ attempts })).toBe(54)
  attempts.push({ id: 'in-flight', advanced: false,
    usage: { prompt_tokens: 10, completion_tokens: 1, cache_read_tokens: 0, cache_creation_tokens: 0, total_tokens: 11 } })
  expect(consumedReviewTokens({ attempts })).toBe(65)
  expect(nextReviewTokenReserve({ attempts })).toBe(0)
  expect(nextReviewTokenReserve({ attempts: [] })).toBe(0)
})

test('novel verification proceeds but two wholly redundant rounds stop', () => {
  const action: SingleReviewAction = { action: 'tools', rationale: 'Check evidence',
    calls: [{ tool: 'read', path: 'docs/GDD.md', startLine: 1, endLine: 1 }] }
  const transcript = [{ action, feedback: 'original line' }]
  expect(reviewProgressStalled(transcript, action)).toBe(false)
  transcript.push({ action, feedback: 'same line' })
  expect(reviewProgressStalled(transcript, action)).toBe(true)
  expect(reviewProgressStalled(transcript, { ...action, calls: [{ tool: 'read', path: 'docs/GDD.md', startLine: 2, endLine: 2 }] })).toBe(false)
})

test('three-message live regression forecasts the last request without changing cumulative charges', () => {
  const attempts: SingleReviewAttempt[] = [{ id: 'live-shaped', advanced: true, closed: {
    kind: 'closed', attemptId: 'live-shaped', usageObserved: true,
    usage: { prompt_tokens: 77464, cache_read_tokens: 149928, cache_creation_tokens: 0, completion_tokens: 9059, total_tokens: 236451 },
    summary: { modelMessages: 3, structuredOutputCalls: 3, structuredOutputFailures: 2, structuredOutputSuccesses: 1, usageSource: 'terminal',
      requests: [{ sequence: 1, inputTokens: 70925, outputTokens: 7392 }, { sequence: 2, inputTokens: 77784, outputTokens: 655 },
        { sequence: 3, inputTokens: 78683, outputTokens: 1012 }] },
    outcome: { kind: 'failed', code: 'fixture' },
  } }]
  expect(consumedReviewTokens({ attempts })).toBe(236451)
  expect(nextReviewTokenReserve({ attempts })).toBe(82779)
  const transcript = [{ action: { action: 'tools' as const, rationale: 'Check', calls: [{ tool: 'calculate' as const, expression: 1 }] }, feedback: 'new host feedback' }]
  expect(nextReviewTokenReserve({ attempts, transcript })).toBeGreaterThan(82779)
  expect(consumedReviewTokens({ attempts }) + nextReviewTokenReserve({ attempts })).toBeLessThan(400000)
  expect(reviewCompletionForecast({ attempts, transcript: [] })).toEqual({
    nextRequest: 82779, minimumRequests: 1, completion: 82779, withOneCorrection: 165558 })
  attempts[0]!.closed!.summary!.requests![2]!.inputTokens = null
  expect(nextReviewTokenReserve({ attempts })).toBe(0)
  delete attempts[0]!.closed!.summary!.requests
  expect(nextReviewTokenReserve({ attempts })).toBe(0)
  attempts[0]!.closed!.summary!.modelMessages = 1
  expect(nextReviewTokenReserve({ attempts })).toBe(231488)
})
