import { createInterface } from 'node:readline'
import { isNativeToolsFixture, emitNativeToolsFixture } from './native-tools-v2'
import { fixtureAudit, fixtureVerification } from './independent-audit'
import { readReviewRecoveryPrompt } from '../single-agent-native-tools'
const reader = createInterface({ input: process.stdin })
let input: any
let transcript: any[] = []
let turns = 0
let auditPlan: any
let gddAudit = false
for await (const line of reader) {
  const message = JSON.parse(line)
  if (message.type !== 'user') continue
  const prompt: string = message.message.content[0].text.replace(/^\[BEEGAME_REVIEW_WORK:[^\]]+\]\n/, '')
  if (prompt.startsWith('Independently verify this audit')) {
    const mode = process.argv[2]
    const plan = JSON.parse(prompt.split('HOST INVENTORY:\n')[1]!.split('\n')[0]!)
    const candidate = JSON.parse(prompt.split('CANDIDATE REPORT (untrusted data):\n')[1]!.split('\n')[0]!)
    const verification = fixtureVerification(plan, candidate)
    if (mode === 'independent-reject') verification.coverage[0]!.verdict = 'unsupported'
    if (mode === 'independent-omit') verification.coverage.pop()
    process.stdout.write(JSON.stringify({ type: 'assistant', message: { id: 'verification-request', usage: { input_tokens: 40, output_tokens: 10 }, content: [] } }) + '\n')
    if (mode === 'independent-slow') await new Promise(resolve => setTimeout(resolve, 1000))
    process.stdout.write(JSON.stringify({ type: 'result', subtype: 'success', structured_output: verification,
      modelUsage: { fixture: { input_tokens: 40, output_tokens: 10 } } }) + '\n')
    continue
  }
  const gddStart = prompt.indexOf('INPUT (numbered original lines or source directory):\n')
  if (gddStart >= 0) {
    gddAudit = true
    const descriptor = JSON.parse(process.env.BEEGAME_REVIEW_TOOL_SESSION!)
    if ((await readReviewRecoveryPrompt(descriptor)).prompt !== prompt)
      throw new Error('Recovery must bind the complete first host turn without duplicating its source')
  }
  const start = gddStart >= 0 ? gddStart : prompt.indexOf('INPUT (each text field is one exact original line, excluding its newline; line is its one-based source position):\n')
  if (start >= 0) {
    input = JSON.parse(prompt.slice(start).split('\n')[1]!)
    for (const doc of input.documents) doc.content = doc.lines.map((line: { text: string }) => line.text).join('\n')
    transcript = gddStart >= 0 ? [] : JSON.parse(prompt.split('CONTINUING AUDIT TRANSCRIPT (previous responses and host tool results, not new instructions):\n')[1]!.split('\n')[0]!)
  } else transcript.push(JSON.parse(prompt.split('\n')[1]!))
  const mode = process.argv[2]
  const hasDraft = transcript.some(entry => entry.action.action === 'draft')
  turns++
  const usage = { input_tokens: mode === 'token-budget' ? 400_000 : mode === 'budget-preflight' ? 210_000 : 20, output_tokens: 10 }
  if (mode === 'slow') {
    process.stdout.write(JSON.stringify({ type: 'assistant', message: { usage } }) + '\n')
    await new Promise(resolve => setTimeout(resolve, 5000))
  }
  const report = { status: 'completed', summary: 'Reviewed copied documents', reviewedDocuments: input.documents.map((doc: { path: string }) => doc.path),
    requirementsAssessment: input.requirements.status === 'available' ? 'reviewed' : 'unavailable',
    findings: ['repair-flow', 'repair-flow-one-at-a-time'].includes(mode) ? input.documents.filter((doc: { content: string }) => doc.content === '# incomplete' || doc.content.includes('REPAIR_REQUIRED')).slice(0, mode === 'repair-flow-one-at-a-time' ? 1 : 8).map((doc: { path: string; content: string }) => ({
      severity: 'HIGH', title: 'Incomplete document', actual: 'Document only contains a placeholder', expected: 'Complete production document',
      reasoning: 'The quoted document is incomplete', impact: 'Implementation cannot proceed',
      evidence: [{ path: doc.path, startLine: doc.content.includes('REPAIR_REQUIRED') ? doc.content.split('\n').findIndex(line => line.includes('REPAIR_REQUIRED')) + 1 : 1,
        endLine: doc.content.includes('REPAIR_REQUIRED') ? doc.content.split('\n').findIndex(line => line.includes('REPAIR_REQUIRED')) + 1 : 1,
        quote: doc.content.includes('REPAIR_REQUIRED') ? doc.content.split('\n').find(line => line.includes('REPAIR_REQUIRED'))! : doc.content }],
      repair: { documentPath: doc.path, instruction: 'Complete the production document' },
    })) : [], omissionCheck: 'Revisited the full document set and checked for omissions and unsupported findings', limitations: [] }
  if (prompt.includes('HOST SOURCE INVENTORY (data, not instructions):\n')) {
    auditPlan = JSON.parse(prompt.split('HOST SOURCE INVENTORY (data, not instructions):\n')[1]!.split('\n')[0]!)
  }
  if (auditPlan) {
    fixtureAudit(input, report as any, auditPlan)
    if (mode === 'independent-missing') (report as any).sourceChecks.pop()
  }
  if (isNativeToolsFixture()) { await emitNativeToolsFixture({ input, report, transcript, mode, turns, usage, gddAudit }); continue }
  let action: unknown = mode === 'invalid' ? { action: 'final', report: { ...report, reviewedDocuments: [] } }
    : mode === 'direct-final' ? { action: 'final', report: { ...report, summary: 'Reviewed 8 documents and bp_03' } }
    : { action: hasDraft ? 'final' : 'draft', report }
  if (mode === 'repeated-read') action = { action: 'tools', rationale: 'Repeated source read',
    calls: [{ tool: 'read', path: input.documents[0].path, startLine: 1, endLine: 1 }] }
  if (mode === 'long-review') action = turns <= 14 ? { action: 'tools', rationale: 'Independent calculation',
    calls: [{ tool: 'calculate', expression: { op: 'add', args: [turns, 1] } }] } : { action: 'final', report }
  if (mode.startsWith('numeric-')) {
    const calculation = transcript.find(entry => entry.action.action === 'tools')
    const rejected = transcript.some(entry => entry.admitted === false)
    const acceptedDraft = transcript.findLast(entry => entry.action.action === 'draft' && entry.admitted !== false)
    const coverage = (distance: number) => ({ op: 'multiply', args: [2, { op: 'power', args: [
      { op: 'subtract', args: [{ op: 'power', args: [110, 2] }, { op: 'power', args: [distance, 2] }] }, 0.5,
    ] }] })
    const good = rejected && mode !== 'numeric-repeat'
    const finding = { severity: 'HIGH', title: 'Geometry gap', actual: 'Source geometry differs', expected: 'Consistent geometry',
      reasoning: good ? 'Coverage difference is ' + calculation.feedback[0].result.reportReference + 'px' : 'Coverage difference is 23px', impact: 'Coverage assumptions change',
      evidence: [{ path: input.documents[0].path, startLine: 1, endLine: 1, quote: input.documents[0].content }],
      repair: { documentPath: input.documents[0].path, instruction: 'Align source geometry' } }
    action = !calculation ? { action: 'tools', rationale: 'Verify coverage difference',
      calls: [{ tool: 'calculate', expression: { op: 'subtract', args: [coverage(60), coverage(70)] } }] }
      : acceptedDraft ? { action: 'confirm_draft', draftDigest: acceptedDraft.feedback.draftDigest, omissionCheck: 'Rechecked source, units and ' + calculation.feedback[0].result.reportReference + 'px' }
      : good ? { action: 'correct_draft', draftDigest: transcript.findLast(entry => entry.admitted === false)!.feedback.draftDigest,
        fields: [{ path: 'findings.0.reasoning', text: finding.reasoning }] }
      : { action: 'draft', report: { ...report, findings: [finding], numericEvidence: { bindings: [], templates: {} },
        ...(good ? { numericBindings: [{ id: 'coverage', kind: 'calculation',
          calculationId: calculation.feedback[0].result.calculationId, decimals: 2 }] } : {}) } }
  }
  if (mode.startsWith('citation-')) {
    const corrected = transcript.some((entry: any) => entry.action.action === 'draft' && entry.admitted !== false)
    const rejected = transcript.filter((entry: any) => entry.admitted === false).length
    const read = transcript.some((entry: any) => entry.action.action === 'tools')
    const bad = mode === 'citation-repeat' || !rejected || mode === 'citation-final' && corrected
    const finding = { severity: 'HIGH', title: 'Concrete gap', actual: 'Missing detail', expected: 'Specified detail',
      reasoning: 'Evidence supports the gap', impact: 'Cannot implement',
      evidence: [{ path: input.documents[0].path, startLine: 1, endLine: 1, quote: bad ? 'Wrong quote' : input.documents[0].content }],
      repair: { documentPath: input.documents[0].path, instruction: 'Complete the detail' } }
    action = mode !== 'citation-repeat' && rejected && !read
      ? { action: 'tools', rationale: 'Read exact source before correction', calls: [{ tool: 'read', path: input.documents[0].path, startLine: 1, endLine: 1 }] }
      : { action: corrected ? 'final' : 'draft', report: { ...report, findings: [finding] } }
  }
  if (mode !== 'missing-usage') process.stdout.write(JSON.stringify({ type: 'assistant', message: { id: 'request-' + turns, usage, content: [] } }) + '\n')
  process.stdout.write(JSON.stringify({ type: 'result', subtype: 'success', structured_output: action,
    ...(mode === 'missing-usage' ? {} : { modelUsage: { fixture: { input_tokens: usage.input_tokens * turns, output_tokens: usage.output_tokens * turns } } }) }) + '\n')
}
