import { expect, test } from 'bun:test'
import { readFileSync } from 'node:fs'
import { fileURLToPath } from 'node:url'
import { ReviewAdmissionError } from '../single-agent-diagnostic'
import { REVIEW_DOCUMENT_PATHS, captureSingleReviewInput, reviewHash, type SingleReviewReport } from '../single-agent-contract'
import { admitNativeReviewReport } from '../single-agent-report'
import { createNativeToolReviewTask } from '../single-agent-tool-task'
import { createContinuingSingleReviewTask } from '../single-agent-task'

function input(large = false) {
  return captureSingleReviewInput({
    subject: { kind: 'initial', sourceResultId: 'shared-report', sourceDocumentRevision: 'a'.repeat(64), confirmedBriefDigest: reviewHash('Brief') },
    requirements: { status: 'unavailable', reason: 'Fixture has no requirements' },
    documents: REVIEW_DOCUMENT_PATHS.map(path => ({ path, content: large ? '# Source\n' + 'source '.repeat(7000) : '# Source\nExact source' })),
  })
}

function report(): SingleReviewReport {
  return {
    status: 'completed', summary: 'All copied sources were reviewed', reviewedDocuments: [...REVIEW_DOCUMENT_PATHS],
    requirementsAssessment: 'unavailable', findings: [], omissionCheck: 'Checked omissions and limitations',
    limitations: ['Requirements were unavailable'],
  }
}

function findingReport(): SingleReviewReport {
  return {
    ...report(), findings: [{ severity: 'HIGH', title: 'Bounded issue', actual: 'Observed', expected: 'Required',
      reasoning: 'Compared source bytes', impact: 'Delivery risk',
      evidence: [{ path: 'docs/GDD.md', startLine: 2, endLine: 2, quote: 'Exact source' }],
      repair: { documentPath: 'docs/GDD.md', instruction: 'Repair the issue' } }],
  }
}

test('legacy and native final admission share citation materialization and outcome', () => {
  const source = input()
  const value = findingReport()
  delete (value.findings[0]!.evidence[0] as { quote?: string }).quote
  const legacy = createContinuingSingleReviewTask(source, []).admit({ action: 'final', report: value })
  const native = createNativeToolReviewTask(source, []).admit(value)
  expect(legacy).toEqual(native)
  expect(native).toMatchObject({ kind: 'accepted', action: { action: 'final', report: findingReport() } })
})

test('the shared native gate rejects invalid ranges without clamping or inventing evidence', () => {
  const value = findingReport()
  delete (value.findings[0]!.evidence[0] as { quote?: string }).quote
  value.findings[0]!.evidence[0]!.endLine = 999
  const legacy = createContinuingSingleReviewTask(input(), []).admit({ action: 'final', report: value })
  const native = createNativeToolReviewTask(input(), []).admit(value)
  expect(legacy).toEqual(native)
  expect(native).toMatchObject({ kind: 'rejected', diagnostic: { reason: 'citation_mismatch', findingIndex: 0, citationIndex: 0 } })
  if (native.kind !== 'rejected') throw new Error('Expected citation rejection')
  expect(native.action.report.findings[0]!.evidence[0]!.quote).toBe('[invalid source range]')
})

test('final requires complete paged reads while draft admission does not', () => {
  const source = input(true)
  const value = report()
  const final = createNativeToolReviewTask(source, []).admit(value)
  expect(final).toMatchObject({ kind: 'rejected', diagnostic: { reason: 'source_not_read' } })
  expect(admitNativeReviewReport(source, value, [], 'draft')).toMatchObject({ kind: 'accepted', action: { action: 'draft' } })
})

test('native and legacy paths retain strict numeric-placeholder rejection', () => {
  const value = report()
  value.summary = 'Unverified distance {{q:missing}}'
  const legacy = createContinuingSingleReviewTask(input(), []).admit({ action: 'final', report: value })
  const native = createNativeToolReviewTask(input(), []).admit(value)
  expect(legacy).toEqual(native)
  expect(native).toMatchObject({ kind: 'rejected', diagnostic: { reason: 'numeric_evidence_invalid' } })
})

test('shared report admission does not mutate input, transcript, or report arguments', () => {
  const source = input()
  const history = [{ action: { action: 'tools' as const, rationale: 'read', calls: [{ tool: 'read' as const, path: 'docs/GDD.md', startLine: 1, endLine: 2 }] },
    feedback: [{ call: { tool: 'read' as const, path: 'docs/GDD.md', startLine: 1, endLine: 2 }, result: { path: 'docs/GDD.md', startLine: 1, endLine: 2, content: '# Source\nExact source' } }] }]
  const originalSource = structuredClone(source)
  const originalHistory = structuredClone(history)
  const value = findingReport()
  delete (value.findings[0]!.evidence[0] as { quote?: string }).quote
  const originalValue = structuredClone(value)
  const task = createNativeToolReviewTask(source, history)
  expect(task.admit(value)).toMatchObject({ kind: 'accepted' })
  expect(source).toEqual(originalSource)
  expect(history).toEqual(originalHistory)
  expect(value).toEqual(originalValue)
})

test('the shared admission keeps the 2 MiB output guard before report parsing', () => {
  const text = 'x'.repeat(2000)
  const oversized = { ...report(), findings: Array.from({ length: 200 }, (_, index) => ({ severity: 'LOW' as const,
    title: text + index, actual: text, expected: text, reasoning: text, impact: text,
    evidence: [{ path: 'docs/GDD.md' as const, startLine: 1, endLine: 1 }],
    repair: { documentPath: 'docs/GDD.md' as const, instruction: text } })) }
  let error: unknown
  try { admitNativeReviewReport(input(), oversized, [], 'final') }
  catch (caught) { error = caught }
  expect(error).toBeInstanceOf(ReviewAdmissionError)
  expect((error as ReviewAdmissionError).diagnostic.reason).toBe('output_too_large')
})

test('legacy action schema keeps its path union while v2 uses the plain enum schema', () => {
  const source = input()
  const legacySchema = JSON.stringify(createContinuingSingleReviewTask(source, []).schema)
  const nativeSchema = JSON.stringify(createNativeToolReviewTask(source, []).schema)
  expect(legacySchema).toContain('"path":{"anyOf"')
  expect(nativeSchema).not.toContain('"path":{"anyOf"')
})

test('bundled v2 task and tools do not load the legacy action adapter or model transport', async () => {
  async function dependencies(entry: string) {
    const loaded = new Set<string>()
    const result = await Bun.build({ entrypoints: [fileURLToPath(new URL(entry, import.meta.url))], target: 'bun',
      plugins: [{ name: 'record-review-dependencies', setup(build) {
        build.onLoad({ filter: /\.[cm]?[jt]sx?$/ }, async args => {
          loaded.add(args.path)
          const loader = args.path.endsWith('.tsx') ? 'tsx' : args.path.endsWith('.ts') ? 'ts' : args.path.endsWith('.jsx') ? 'jsx' : 'js'
          return { contents: await Bun.file(args.path).text(), loader }
        })
      } }] })
    if (!result.success) throw new Error(result.logs.map(log => log.message).join('\n'))
    expect([...loaded].some(path => path.endsWith(entry.slice(3)))).toBe(true)
    expect([...loaded].some(path => path.endsWith('single-agent-report.ts'))).toBe(true)
    return [...loaded]
  }
  for (const entry of ['../single-agent-tool-task.ts', '../single-agent-native-tools.ts']) {
    const loaded = (await dependencies(entry)).join('\n')
    expect(loaded).not.toMatch(/single-agent-task\.ts|single-agent-model-format\.ts/)
  }
})

test('v2 report/task modules have no runtime dependency on the legacy action adapter source', () => {
  const files = ['../single-agent-report.ts', '../single-agent-tool-task.ts', '../single-agent-native-tools.ts']
    .map(path => readFileSync(fileURLToPath(new URL(path, import.meta.url)), 'utf8'))
  for (const source of files) expect(source).not.toMatch(/from ['"]\.\/single-agent-task['"]|require\(['"]\.\/single-agent-task['"]\)/)
})
