function fixture() {
  const input: SingleReviewInput = { subject: { kind: 'initial', sourceResultId: 'source', sourceDocumentRevision: reviewHash('source'), confirmedBriefDigest: reviewHash('Required capacity: 10') },
    requirements: { status: 'available', content: 'Required capacity: 10', digest: reviewHash('Required capacity: 10') },
    documents: REVIEW_DOCUMENT_PATHS.map(path => ({ path, content: '# Specification\nCapacity: 5\n## Constraints\nIndependent constraint.' })) }
  const report: SingleReviewReport = { status: 'completed', summary: 'Fixture audit', reviewedDocuments: [...REVIEW_DOCUMENT_PATHS], requirementsAssessment: 'reviewed',
    findings: [], omissionCheck: 'Fixture checked omissions', limitations: [] }
  return { input, report: fixtureAudit(input, report) }
}
function withFinding() {
  const { input, report } = fixture()
  report.findings.push({ severity: 'HIGH', title: 'Capacity mismatch', actual: 'Capacity 5', expected: 'Capacity 10', reasoning: 'Same scope differs', impact: 'Cannot satisfy requirement',
    evidence: [{ path: 'docs/GDD.md', startLine: 2, endLine: 2, quote: 'Capacity: 5' }], repair: { documentPath: 'docs/GDD.md', instruction: 'Align capacity' } })
  return { input, report: fixtureAudit(input, report) }
}
test('inventory covers all nonblank original lines, including fenced blocks, and changes when source changes', () => {
  const { input } = fixture()
  input.documents[0]!.content = '# A\n' + Array.from({ length: 65 }, (_, i) => 'row ' + i).join('\n') + '\n```\n# not a heading\n```'
  const plan = buildIndependentAuditPlan(input)
  for (const doc of input.documents) for (const [index, line] of doc.content.split('\n').entries())
    if (line.trim()) expect(plan.units.filter(unit => unit.path === doc.path && unit.startLine <= index + 1 && unit.endLine >= index + 1)).toHaveLength(1)
  expect(plan.units.every(unit => unit.endLine - unit.startLine < 24)).toBe(true)
  input.documents[3]!.content += '\nnew independent constraint'
  expect(buildIndependentAuditPlan(input).digest).not.toBe(plan.digest)
})
test('mechanical links and explicit formula namespaces are host findings, not self-reported omissions', () => {
  const { input, report } = fixture()
  input.documents[0]!.content = '# Links\n[docs/UI_UX_SPEC.md](docs/ART_DIRECTION.md)\n[missing](MISSING.md)\n[external](https://example.org/no.md)\n`[code](NO.md)`\n```\n[code](NO.md)\n```\n| F_RATE | defined |\nUse F_RATE and F_RTAE.'
  const plan = buildIndependentAuditPlan(input)
  expect(plan.mechanicalChecks.map(m => m.kind).sort()).toEqual(['document_label_target_mismatch', 'document_target_missing', 'formula_reference_undefined'])
  expect(plan.units[0]!.dependencies).toEqual(['docs/ART_DIRECTION.md'])
  const canonical = appendMechanicalFindings(input, fixtureAudit(input, report))
  assertMechanicalFindings(input, canonical)
  canonical.findings.pop()
  expect(() => assertMechanicalFindings(input, canonical)).toThrow()
})
test('whole-audit assertions cannot substitute for per-unit and dependency evidence', () => {
  const { input, report } = fixture()
  assertIndependentAudit(input, report, [])
  report.sourceChecks!.pop()
  expect(independentAuditProblems(input, report, []).some(p => p.code === 'missing_unit')).toBe(true)
  input.documents[0]!.content += '\n[constraint](TECHNICAL_DESIGN.md)'
  fixtureAudit(input, report)
  const check = report.sourceChecks!.find(c => c.evidence.some(e => e.path === 'docs/TECHNICAL_DESIGN.md') && c.evidence[0]!.path === 'docs/GDD.md')!
  check.evidence.pop()
  expect(independentAuditProblems(input, report, []).some(p => p.code === 'dependency_not_checked')).toBe(true)
})

test('fabricated premises, unresolved checks, duplicates and foreign unit IDs fail closed', () => {
  const { input, report } = withFinding()
  assertIndependentAudit(input, report, [])
  report.findings[0]!.evidenceProof!.premises[0]!.excerpt = 'Fabricated source text'
  report.sourceChecks![0]!.disposition = 'uncertain'
  report.sourceChecks!.push(report.sourceChecks![0]!)
  report.sourceChecks![1]!.unitId = 'unit-' + 'f'.repeat(64)
  const codes = independentAuditProblems(input, report, []).map(p => p.code)
  for (const code of ['premise_not_source', 'unresolved_check', 'duplicate_unit', 'unknown_unit', 'missing_unit'] as const) expect(codes).toContain(code)
})
test('calculation proof requires an executed matching successful host result', () => {
  const { input, report } = withFinding()
  const call = { tool: 'calculate' as const, expression: { op: 'add' as const, args: [2, 3] } }
  const result = executeSingleReviewTool(input, call) as { calculationId: string; value: number }
  const transcript: SingleReviewTranscriptEntry[] = [{ action: { action: 'tools', rationale: 'check arithmetic', calls: [call] }, feedback: [{ call, result }] }]
  report.findings[0]!.evidenceProof!.calculations = [{ calculationId: result.calculationId, value: result.value }]
  expect(independentAuditProblems(input, report, []).some(p => p.code === 'calculation_not_executed')).toBe(true)
  assertIndependentAudit(input, report, transcript)
  ;(transcript[0]!.feedback as any)[0].result.value = 999
  expect(auditedCalculations(input, transcript)).toEqual([])
  expect(independentAuditProblems(input, report, transcript).some(p => p.code === 'calculation_not_executed')).toBe(true)
})
test('verifier has fresh context, source-bound verdicts and exact candidate binding', () => {
  const { input, report } = withFinding(), plan = buildIndependentAuditPlan(input)
  const verdict = fixtureVerification(plan, report)
  const task = createIndependentVerificationTask(input, report, [{ action: { action: 'final', report }, feedback: 'SECRET AUDITOR SELF ASSURANCE' }])
  expect(task.nativeTools.transcript).toEqual([])
  expect(task.prompt).not.toContain('SECRET AUDITOR SELF ASSURANCE')
  expect(task.admit(verdict)).toEqual(verdict)
  verdict.coverage.pop()
  expect(() => task.admit(verdict)).toThrow('coverage')
  const tampered = fixtureVerification(plan, report); tampered.reportDigest = reviewDigest('different')
  expect(() => task.admit(tampered)).toThrow('binding')
})
test('unsupported, uncertain and omitted semantic checks cannot authorize publication', () => {
  const { input, report } = withFinding(), plan = buildIndependentAuditPlan(input)
  const start = { protocolVersion: 1 as const, ownerId: 'owner', projectId: 'project', requestId: 'request', input,
    inputDigest: singleReviewInputDigest({ ownerId: 'owner', projectId: 'project', input }), validationPolicy: INDEPENDENT_REVIEW_POLICY }
  expect(() => buildSingleReviewResult(start, report)).toThrow()
  for (const verdict of ['unsupported', 'uncertain'] as const) {
    const verification = fixtureVerification(plan, report); verification.findings[0]!.verdict = verdict
    validateIndependentVerification(input, report, verification)
    expect(independentVerificationPassed(verification)).toBe(false)
    expect(() => buildSingleReviewResult(start, report, verification)).toThrow()
  }
  const verification = fixtureVerification(plan, report)
  verification.omissions.push({ unitId: plan.units[0]!.id, description: 'Unreported conflict', evidence: verification.coverage[0]!.evidence })
  expect(() => buildSingleReviewResult(start, report, verification)).toThrow()
  const published = buildSingleReviewResult(start, report, fixtureVerification(plan, report))
  expect(published.audit.validation!.sourceChecks).toHaveLength(plan.units.length)
})
test('native production schema requires provenance and does not permit model-owned mechanical findings', () => {
  const { input, report } = withFinding()
  const task = createNativeToolReviewTask(input, [], true)
  const native = structuredClone(report) as any
  for (const f of native.findings) for (const e of f.evidence) delete e.quote
  expect(task.admit(native).kind).toBe('accepted')
  delete native.sourceChecks
  expect(() => task.validateStructuredOutput(native)).toThrow()
  native.sourceChecks = report.sourceChecks; native.findings[0].mechanicalCheckId = 'mechanical-' + 'a'.repeat(64)
  expect(() => task.validateStructuredOutput(native)).toThrow()
})
test('independent verifier cannot inherit auditor paged-read coverage', () => {
  const { input, report } = fixture()
  for (const doc of input.documents) doc.content = 'Original '.repeat(5000)
  fixtureAudit(input, report)
  const transcript: SingleReviewTranscriptEntry[] = [...input.documents, { path: 'requirements', content: 'Required capacity: 10' }].map(doc => {
    const call = { tool: 'read' as const, path: doc.path as any, startLine: 1, endLine: 1 }
    return { action: { action: 'tools', rationale: 'Read source', calls: [call] }, feedback: [{ call, result: executeSingleReviewTool(input, call) }] }
  })
  const task = createIndependentVerificationTask(input, report, transcript)
  const verification = fixtureVerification(buildIndependentAuditPlan(input), report)
  expect(() => task.admit(verification)).toThrow()
  expect(() => task.admitWithTools(verification, [])).toThrow()
  expect(task.admitWithTools(verification, transcript)).toEqual(verification)
})
