import { z } from 'zod/v4'
import { reviewCitationSchema, singleReviewReportSchema, reviewDigest, type SingleReviewInput } from './single-agent-contract'
import { buildSingleReviewPrompt } from './single-agent-prompt'
import { ReviewAdmissionError, type ReviewOutputDiagnostic } from './single-agent-diagnostic'
import { reviewNumericBindingsSchema, type ReviewNumericBinding } from './single-agent-numeric-schema'
import {
  admitCanonicalReviewReport,
  archivedSingleReviewActionSchema,
  legacyNativeReviewReportSchema,
  normalizeReviewReport,
  referenceReviewReportSchema,
  singleReviewActionSchema,
  type SingleReviewAction,
  type SingleReviewTranscriptEntry,
} from './single-agent-report'
// Compatibility exports for legacy callers and journal readers.  The v2
// task/tools modules import the report module directly and do not load this
// legacy action adapter.
export {
  archivedSingleReviewActionSchema,
  legacyNativeReviewReportSchema,
  nativeReviewReportSchema,
  singleReviewActionSchema,
} from './single-agent-report'
export type { SingleReviewAction, SingleReviewTranscriptEntry } from './single-agent-report'

export type SingleReviewDecision = { kind: 'accepted'; action: SingleReviewAction } |
  { kind: 'rejected'; action: SingleReviewAction; diagnostic: ReviewOutputDiagnostic & { stage: 'admission'; reason: 'citation_mismatch' | 'numeric_evidence_invalid' | 'source_not_read' | 'independent_audit_invalid' } }

/** Retain only schema-bounded citation rejections for the same auditor to correct. */
export function createContinuingSingleReviewTask(input: SingleReviewInput, transcript: readonly SingleReviewTranscriptEntry[]) {
  const task = createSingleReviewTask(input, transcript)
  function decide(value: unknown, replay: boolean): SingleReviewDecision {
    try { return { kind: 'accepted', action: replay ? task.replay(value) : task.admit(value) } }
    catch (error) {
      if (!(error instanceof ReviewAdmissionError) ||
        (error.diagnostic.reason !== 'citation_mismatch' && error.diagnostic.reason !== 'numeric_evidence_invalid' && error.diagnostic.reason !== 'source_not_read')) throw error
      return { kind: 'rejected', action: normalizeAction(input, value, transcript, false, replay),
        diagnostic: { ...error.diagnostic, stage: 'admission', reason: error.diagnostic.reason } }
    }
  }
  return { format: task.format, validateStructuredOutput: task.validateStructuredOutput, prompt: task.prompt, schema: task.schema,
    admit: (value: unknown) => decide(value, false), replay: (value: unknown) => decide(value, true) }
}

// The model selects source positions; the host copies the evidence bytes.
// Supplied legacy quotes remain strictly verified, never silently repaired.
const referenceAction = z.discriminatedUnion('action', [
  singleReviewActionSchema.options[0],
  z.object({ action: z.literal('draft'), report: referenceReviewReportSchema }).strict(),
  z.object({ action: z.literal('final'), report: referenceReviewReportSchema }).strict(),
  z.object({ action: z.literal('amend_draft'), draftDigest: z.string().regex(/^[a-f0-9]{64}$/),
    citations: z.array(z.object({
      findingIndex: z.number().int().nonnegative().safe(), citationIndex: z.number().int().nonnegative().safe(),
      reference: reviewCitationSchema.omit({ quote: true }),
    }).strict()).min(1).max(6400),
  }).strict(),
  z.object({ action: z.literal('confirm_draft'), draftDigest: z.string().regex(/^[a-f0-9]{64}$/),
    omissionCheck: singleReviewReportSchema.shape.omissionCheck,
    numericBindings: reviewNumericBindingsSchema.optional(),
  }).strict(),
  z.object({ action: z.literal('correct_draft'), draftDigest: z.string().regex(/^[a-f0-9]{64}$/),
    fields: z.array(z.object({ path: z.string().regex(/^(summary|omissionCheck|limitations\.\d+|findings\.\d+\.(title|actual|expected|reasoning|impact|repair\.instruction))$/),
      text: z.string().min(1).max(8000) }).strict()).max(512),
    numericBindings: reviewNumericBindingsSchema.optional(),
  }).strict(),
])
const nativeAction = z.discriminatedUnion('action', [
  referenceAction.options[0],
  z.object({ action: z.literal('draft'), report: legacyNativeReviewReportSchema }).strict(),
  z.object({ action: z.literal('final'), report: legacyNativeReviewReportSchema }).strict(),
  referenceAction.options[3], referenceAction.options[4].omit({ numericBindings: true }),
])
function normalizeAction(input: SingleReviewInput, value: unknown, transcript: readonly SingleReviewTranscriptEntry[],
  validateNumbers = true, replay = false): SingleReviewAction {
  if (replay && (value as { action?: unknown } | null)?.action === 'tools')
    return archivedSingleReviewActionSchema.parse(value)
  const wire = referenceAction.parse(value)
  let action: z.infer<typeof referenceAction.options[0]> | { action: 'draft' | 'final'; report: z.infer<typeof referenceReviewReportSchema> }
  let numericBindings: ReviewNumericBinding[] = []
  if (wire.action === 'amend_draft' || wire.action === 'confirm_draft' || wire.action === 'correct_draft') {
    const latest = transcript.findLast(entry => entry.action.action !== 'tools')
    if (!latest || latest.action.action === 'tools' || reviewDigest(latest.action.report) !== wire.draftDigest ||
      (wire.action === 'confirm_draft' && (latest.action.action !== 'draft' || latest.admitted === false)))
      throw new ReviewAdmissionError({ stage: 'admission', reason: 'draft_required' })
    const report: z.infer<typeof referenceReviewReportSchema> = structuredClone(latest.action.report)
    if (wire.action === 'amend_draft') {
      const seen = new Set<string>()
      for (const patch of wire.citations) {
        const finding = report.findings[patch.findingIndex]
        const key = patch.findingIndex + ':' + patch.citationIndex
        if (!finding?.evidence[patch.citationIndex] || seen.has(key))
          throw new ReviewAdmissionError({ stage: 'admission', reason: 'citation_mismatch',
            findingIndex: patch.findingIndex, citationIndex: patch.citationIndex })
        seen.add(key)
        finding.evidence[patch.citationIndex] = patch.reference
      }
    } else if (wire.action === 'correct_draft') {
      const setField = (path: string, text: string) => {
        const parts = path.split('.')
        let target: any = report
        for (const part of parts.slice(0, -1)) {
          if (!target || !Object.hasOwn(target, part)) throw new ReviewAdmissionError({ stage: 'admission', reason: 'draft_required' })
          target = target[part]
        }
        const key = parts.at(-1)!
        if (!target || !Object.hasOwn(target, key) || typeof target[key] !== 'string')
          throw new ReviewAdmissionError({ stage: 'admission', reason: 'draft_required' })
        target[key] = text
      }
      // Re-render every retained template if an existing binding is replaced.
      for (const [path, text] of Object.entries(report.numericEvidence?.templates ?? {})) setField(path, text)
      const seen = new Set<string>()
      for (const patch of wire.fields) {
        if (seen.has(patch.path)) throw new ReviewAdmissionError({ stage: 'admission', reason: 'draft_required' })
        seen.add(patch.path)
        setField(patch.path, patch.text)
        if (report.numericEvidence) delete report.numericEvidence.templates[patch.path]
      }
      numericBindings = wire.numericBindings ?? []
      if (report.numericEvidence) {
        const replaced = new Set(numericBindings.map(binding => binding.id))
        const used = new Set(JSON.stringify(report).match(/\{\{q:([a-z][a-z_]*)\}\}/g)?.map(slot => slot.slice(4, -2)))
        report.numericEvidence.bindings = report.numericEvidence.bindings.filter(binding => !replaced.has(binding.id) && used.has(binding.id))
      }
    } else {
      report.omissionCheck = wire.omissionCheck
      if (report.numericEvidence) delete report.numericEvidence.templates.omissionCheck
      numericBindings = wire.numericBindings ?? []
    }
    action = { action: wire.action === 'confirm_draft' ? 'final' : 'draft', report }
  } else {
    action = wire
    if (action.action !== 'tools') {
      numericBindings = action.report.numericBindings ?? []
      delete action.report.numericBindings
    }
  }
  if (action.action !== 'tools')
    action.report = normalizeReviewReport(input, action.report, transcript, { numericBindings, validateNumbers, replay })
  return singleReviewActionSchema.parse(action)
}
export function createSingleReviewTask(input: SingleReviewInput, transcript: readonly SingleReviewTranscriptEntry[]) {
  const captured = structuredClone({ input, transcript })
  function admit(value: unknown, replay: boolean): SingleReviewAction {
    const serialized = JSON.stringify(value)
    if (!serialized || Buffer.byteLength(serialized) > 2 * 1024 * 1024) throw new ReviewAdmissionError({ stage: 'admission', reason: 'output_too_large' })
    const action = normalizeAction(captured.input, value, captured.transcript, true, replay)
    if (action.action !== 'tools') admitCanonicalReviewReport(captured.input, action.report, captured.transcript, action.action === 'final')
    return action
  }
  return {
    format: 'foundation-review-action' as const,
    validateStructuredOutput: (value: unknown) => { nativeAction.parse(value) },
    prompt: buildSingleReviewPrompt(captured.input, captured.transcript),
    // The native StructuredOutput tool uses the default Ajv draft-7 instance.
    // Its unsupported-dialect path omits the tool rather than failing startup.
    schema: z.toJSONSchema(nativeAction, { target: 'draft-7' }),
    admit: (value: unknown) => admit(value, false),
    // Existing immutable reports stay readable; new numeric proofs are revalidated on replay.
    replay: (value: unknown) => admit(value, true),
  }
}
